    77→export default function BusinessProfileView() {

    85→  const load = useCallback(async () => {
    86→    setLoading(true);
    87→    try {
    88→      const res = await fetch("/api/profile", { cache: "no-store" });
    89→      const data = await res.json();
    90→      if (!res.ok) throw new Error(data.error || "Gagal memuat profil");
    91→      const p: Profile = data.profile;
    92→      setProfile(p);
    93→      setForm({
    94→        businessName: p.businessName ?? "",
    95→        ownerName: p.ownerName ?? "",
    96→        category: p.category ?? "",
    97→        description: p.description ?? "",
    98→        address: p.address ?? "",
    99→        lat: String(p.lat ?? ""),
   100→        lng: String(p.lng ?? ""),
   101→        email: p.email ?? "",
   102→        website: p.website ?? "",
   103→        instagram: p.instagram ?? "",
   104→        facebook: p.facebook ?? "",
   105→        tiktok: p.tiktok ?? "",
   106→        logoUrl: p.logoUrl || "/images/logo.png",
   107→        coverUrl: p.coverUrl ?? "",
   108→      });
   109→      setHoursRows(parseHours(p.hours));
   110→    } catch (e) {
   111→      toast({
   112→        title: "Gagal memuat profil",
   113→        description: e instanceof Error ? e.message : "Coba muat ulang halaman",
   114→        variant: "destructive",
   115→      });
   116→    } finally {
   117→      setLoading(false);
   118→    }
   119→  }, [toast]);
   120→
   121→  useEffect(() => {
   122→    load();
   123→  }, [load]);
   124→
   125→  const patch = (p: Partial<FormState>) => setForm((prev) => ({ ...prev, ...p }));
   126→
   127→  const handleLogoFile = async (file: File) => {
   128→    try {
   129→      // Batch 4: logo diunggah ke server (URL), bukan base64 di DB
   130→      patch({ logoUrl: await uploadImage(file, "logo") });
   131→      toast({ title: "Logo diperbarui", description: "Jangan lupa simpan perubahan." });
   132→    } catch (e) {
   133→      toast({
   134→        title: "Gagal memuat logo",
   135→        description: e instanceof Error ? e.message : "Coba gambar lain",
   136→        variant: "destructive",
   137→      });
   138→    }
   139→  };
   140→
   141→  const handleCoverFile = async (file: File) => {
   142→    try {
   143→      // Latar (cover) profil diunggah ke server (URL), bukan base64 di DB
   144→      patch({ coverUrl: await uploadImage(file, "cover") });
   145→      toast({ title: "Latar belakang diperbarui", description: "Jangan lupa simpan perubahan." });
   146→    } catch (e) {
   147→      toast({
   148→        title: "Gagal memuat latar belakang",
   149→        description: e instanceof Error ? e.message : "Coba gambar lain",
   150→        variant: "destructive",
   151→      });
   152→    }
   153→  };
   154→
   155→  const handleSave = async () => {
   156→    if (!form.businessName.trim()) {
   157→      toast({ title: "Nama bisnis wajib diisi", variant: "destructive" });
   158→      return;
   159→    }
   160→    setSaving(true);
   161→    try {
   162→      const res = await fetch("/api/profile", {
   163→        method: "PATCH",
   164→        headers: { "Content-Type": "application/json" },
   165→        body: JSON.stringify({
   166→          businessName: form.businessName.trim(),
   167→          ownerName: form.ownerName.trim(),
   168→          category: form.category.trim(),
   169→          description: form.description,
   170→          address: form.address.trim(),
   171→          lat: parseFloat(form.lat) || 0,
   172→          lng: parseFloat(form.lng) || 0,
   173→          email: form.email.trim(),
   174→          website: form.website.trim(),
   175→          instagram: form.instagram.replace(/^@+/, "").trim(),
   176→          facebook: form.facebook
   177→            .replace(/^https?:\/\/(www\.)?facebook\.com\//i, "")
   178→            .replace(/^\/+/, "")
   179→            .trim(),
   180→          tiktok: form.tiktok.replace(/^@+/, "").trim(),
   181→          hours: JSON.stringify(hoursRows),
   182→          logoUrl: form.logoUrl,
   183→          coverUrl: form.coverUrl,
   184→        }),
   185→      });
   186→      const data = await res.json();
   187→      if (!res.ok) throw new Error(data.error || "Gagal menyimpan profil");
   188→      setProfile(data.profile);
   189→      // Beri tahu seluruh aplikasi (sidebar, header, layar keluar) bahwa profil berubah
   190→      window.dispatchEvent(new CustomEvent("wa:profile-updated"));

   192→    } catch (e) {
   193→      toast({
   194→        title: "Gagal menyimpan",
   195→        description: e instanceof Error ? e.message : "Coba lagi",
   196→        variant: "destructive",
   197→      });
   198→    } finally {
   199→      setSaving(false);
   200→    }
   201→  };
   202→
   203→  const formProps = { form, patch, hoursRows, setHoursRows, phone: profile?.phone ?? "", onSave: handleSave, saving, onLogoFile: handleLogoFile, onCoverFile: handleCoverFile };
   204→  const previewProps = { form, hoursRows, phone: profile?.phone ?? "", verified: profile?.verified ?? false, category: profile?.category ?? "" };
   205→  const isDesktop = useIsDesktop();
   206→
   207→  return (
   208→    <div className="h-full flex flex-col">
   209→      {/* Header view hanya desktop — di mobile dihapus (judul sudah tampil di panel header atas,
   210→          sesuai anotasi screenshot "Hapus Bagian Ini"). */}
   211→      {isDesktop && (
   212→        <ViewHeader icon={Store} title="Profil Bisnis" subtitle="Kelola informasi toko yang dilihat pelanggan" />
   213→      )}
   214→
   215→      {loading ? (
   216→        <div className="flex-1 overflow-y-auto wa-scrollbar p-4 sm:p-6">
   217→          <div className="max-w-3xl mx-auto space-y-4">
   218→            <div className="flex items-center gap-4">
   219→              <Skeleton className="h-20 w-20 rounded-2xl" />
   220→              <div className="space-y-2">
   221→                <Skeleton className="h-4 w-40" />
   222→                <Skeleton className="h-3 w-52" />
   223→              </div>
   224→            </div>
   225→            <Skeleton className="h-12 w-full" />
   226→            <Skeleton className="h-12 w-full" />
   227→            <Skeleton className="h-28 w-full" />
   228→            <Skeleton className="h-64 w-full" />
   229→          </div>
   230→        </div>
   231→      ) : !profile ? (
   232→        <div className="flex-1 flex items-center justify-center text-sm text-muted-foreground">Profil tidak tersedia.</div>

   234→        <div className="flex-1 min-h-0 overflow-y-auto wa-scrollbar">

   237→            <div className="p-4 sm:p-6">
   238→              <div className="mx-auto grid max-w-6xl gap-6 lg:grid-cols-5">
   239→                <div className="lg:col-span-3">
   240→                  <ProfileForm {...formProps} />
   241→                </div>
   242→                <div className="lg:col-span-2">
   243→                  <ProfilePreview {...previewProps} />
   244→                </div>
   245→              </div>
   246→            </div>
   247→          ) : (
   248→            /* Mobile/tablet: Tabs Pratinjau / Edit (Pratinjau dulu, lalu Edit) */
   249→            <div className="p-4 sm:p-6">
   250→              <Tabs defaultValue="edit">
   251→                <TabsList className="grid w-full grid-cols-2 mb-4">

   253→                  <TabsTrigger value="edit">Edit</TabsTrigger>

   255→                <TabsContent value="preview">
   256→                  <ProfilePreview {...previewProps} />
   257→                </TabsContent>
   258→                <TabsContent value="edit">
   259→                  <ProfileForm {...formProps} />
   260→                </TabsContent>
   261→              </Tabs>
   262→            </div>
   263→          )}
   264→        </div>
   265→      )}
   266→    </div>

   499→                    aria-label={`Buka atau tutup pada hari ${row.day}`}
   500→                  />
   501→                  <span className={cn("text-xs w-10", row.closed ? "text-red-500" : "text-emerald-600")}>{row.closed ? "Tutup" : "Buka"}</span>
   502→                </div>
   503→                <div className={cn("flex items-center gap-2 ml-auto transition-opacity", row.closed && "opacity-40")}>
   504→                  <Input
   505→                    type="time"
   506→                    value={row.open}
   507→                    disabled={row.closed}
   508→                    onChange={(e) => setHour(idx, { open: e.target.value })}

   510→                    aria-label={`Jam buka hari ${row.day}`}
   511→                  />
   512→                  <span className="text-xs text-muted-foreground" aria-hidden="true">–</span>
   513→                  <Input
   514→                    type="time"
   515→                    value={row.close}
   516→                    disabled={row.closed}
   517→                    onChange={(e) => setHour(idx, { close: e.target.value })}
   518→                    className="w-[106px] h-8 text-sm"
   519→                    aria-label={`Jam tutup hari ${row.day}`}
   520→                  />
   521→                </div>
   522→              </div>
   523→            ))}
   524→          </div>
   525→        </div>
   526→
   527→        <div className="flex justify-end border-t pt-4">
   528→          <Button onClick={onSave} disabled={saving} className="min-w-36">
   529→            {saving ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : <Save className="h-4 w-4 mr-2" />}
   530→            {saving ? "Menyimpan..." : "Simpan Profil"}
   531→          </Button>
   532→        </div>
   533→      </CardContent>
   534→    </Card>
   535→  );
