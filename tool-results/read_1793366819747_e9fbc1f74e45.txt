    70→/** Bentuk mentah dari API: groups berupa string JSON; flag boleh absen pada respons parsial. */

    77→function parseGroups(raw: string): GroupT[] {
    78→  try {
    79→    const v = JSON.parse(raw);
    80→    return Array.isArray(v) ? (v as GroupT[]) : [];
    81→  } catch {
    82→    return [];
    83→  }
    84→}
    85→
    86→function toCommunity(raw: CommunityRaw): CommunityT {
    87→  return { ...raw, groups: parseGroups(raw.groups), isMine: !!raw.isMine, canManage: !!raw.canManage };
    88→}

    90→function fmtCount(n: number): string {
    91→  try {
    92→    return new Intl.NumberFormat("id-ID", { notation: "compact", maximumFractionDigits: 1 }).format(n);
    93→  } catch {
    94→    return String(n);
    95→  }

   154→/** Dialog Buat/Edit Komunitas — satu form untuk dua mode, lengkap pratinjau avatar langsung (Task 46). */
   155→function CommunityFormDialog({
   156→  community,
   157→  open,
   158→  onOpenChange,
   159→  onSaved,
   160→}: {
   161→  community: CommunityT | null; // null = mode buat
   162→  open: boolean;
   163→  onOpenChange: (open: boolean) => void;
   164→  onSaved: (community: CommunityT) => void;
   165→}) {
   166→  const { toast } = useToast();
   167→  const editing = community !== null;
   168→  const [name, setName] = useState("");
   169→  const [description, setDescription] = useState("");
   170→  const [colorKey, setColorKey] = useState<string>("teal");
   171→  const [saving, setSaving] = useState(false);
   172→
   173→  useEffect(() => {
   174→    if (open) {
   175→      setName(community?.name ?? "");
   176→      setDescription(community?.description ?? "");
   177→      setColorKey(community?.colorKey ?? "teal");
   178→    }
   179→  }, [open, community]);
   180→
   181→  async function save() {
   182→    if (saving) return;
   183→    setSaving(true);
   184→    try {
   185→      const res = await fetch(editing ? `/api/communities/${community.id}` : "/api/communities", {
   186→        method: editing ? "PATCH" : "POST",
   187→        headers: { "Content-Type": "application/json" },
   188→        cache: "no-store",
   189→        body: JSON.stringify({ name: name.trim(), description: description.trim(), colorKey }),
   190→      });
   191→      const data = await res.json();
   192→      if (!res.ok || !data?.community) {
   193→        throw new Error(data?.error || (editing ? "Gagal menyimpan perubahan" : "Gagal membuat komunitas"));
   194→      }
   195→      onSaved(toCommunity(data.community as CommunityRaw));
   196→      toast({
   197→        title: editing ? "Komunitas diperbarui" : "Komunitas dibuat",
   198→        description: editing
   199→          ? `Perubahan pada "${name.trim()}" tersimpan.`
   200→          : `Komunitas "${name.trim()}" tersimpan dan siap digunakan.`,
   201→      });
   202→      onOpenChange(false);
   203→    } catch (e) {
   204→      toast({
   205→        title: editing ? "Gagal menyimpan perubahan" : "Gagal membuat komunitas",
   206→        description: e instanceof Error ? e.message : "Silakan coba lagi.",
   207→        variant: "destructive",
   208→      });
   209→    } finally {
   210→      setSaving(false);
   211→    }
   212→  }
   213→
   214→  return (
   215→    <Dialog open={open} onOpenChange={onOpenChange}>
   216→      <DialogContent className="sm:max-w-md">
   217→        <DialogHeader>
   218→          <DialogTitle className="flex items-center gap-2">
   219→            {editing ? <PencilLine className="h-5 w-5 text-primary" /> : <Users className="h-5 w-5 text-primary" />}
   220→            {editing ? "Edit Komunitas" : "Buat Komunitas"}
   221→          </DialogTitle>
   222→          <DialogDescription>
   223→            {editing
   224→              ? "Perbarui nama, deskripsi, atau warna komunitas."
   225→              : "Satukan beberapa grup terkait bisnis dalam satu komunitas."}
   226→          </DialogDescription>
   227→        </DialogHeader>
   228→        <div className="space-y-3.5 py-1">
   229→          {/* Pratinjau langsung */}
   230→          <div className="flex items-center gap-3 rounded-xl border border-border bg-muted/40 p-3">
   231→            <div
   232→              className={cn(
   233→                "flex h-11 w-11 shrink-0 items-center justify-center overflow-hidden rounded-xl bg-gradient-to-br text-white",
   234→                heroGradient(colorKey)
   235→              )}
   236→              aria-hidden
   237→            >
   238→              {name.trim() ? <span className="text-sm font-bold">{getInitials(name)}</span> : <Users className="h-5 w-5" />}
   239→            </div>
   240→            <div className="min-w-0">
   241→              <p className="truncate text-sm font-semibold">{name.trim() || "Nama komunitas"}</p>
   242→              <p className="text-[11px] text-muted-foreground">Pratinjau avatar komunitas</p>
   243→            </div>
   244→          </div>
   245→          <div className="space-y-1.5">
   246→            <Label htmlFor="community-name">Nama komunitas</Label>
   247→            <Input
   248→              id="community-name"
   249→              value={name}
   250→              onChange={(e) => setName(e.target.value)}
   251→              maxLength={80}
   252→              placeholder="cth. Komunitas Pelanggan IndiGoo"
   253→            />
   254→          </div>
   255→          <div className="space-y-1.5">
   256→            <Label htmlFor="community-desc">Deskripsi</Label>
   257→            <Textarea
   258→              id="community-desc"
   259→              value={description}
   260→              onChange={(e) => setDescription(e.target.value)}
   261→              rows={3}
   262→              maxLength={300}
   263→              placeholder="Untuk apa komunitas ini?"
   264→            />
   265→          </div>
   266→          <div className="space-y-1.5">
   267→            <Label>Warna komunitas</Label>
   268→            <div className="flex flex-wrap gap-2" role="radiogroup" aria-label="Pilih warna komunitas">
   269→              {COLOR_OPTIONS.map((c) => (
   270→                <button
   271→                  key={c}
   272→                  type="button"
   273→                  role="radio"
   274→                  aria-checked={colorKey === c}
   275→                  aria-label={`Warna ${c}`}
   276→                  title={c.charAt(0).toUpperCase() + c.slice(1)}
   277→                  onClick={() => setColorKey(c)}
   278→                  className={cn(
   279→                    "flex h-9 w-9 items-center justify-center rounded-full border-2 transition-all",
   280→                    colorOf(c).bg,
   281→                    colorKey === c
   282→                      ? "scale-110 border-foreground/70 shadow-md"
   283→                      : "border-transparent hover:scale-105 hover:border-border"
   284→                  )}
   285→                >
   286→                  {colorKey === c ? <Check className="h-4 w-4 text-white" aria-hidden /> : null}
   287→                </button>
   288→              ))}
   289→            </div>
   290→          </div>
   291→        </div>
   292→        <DialogFooter>
   293→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={saving}>
   294→            Batal
   295→          </Button>
   296→          <Button onClick={() => void save()} disabled={saving || name.trim().length === 0}>
   297→            {saving ? (
   298→              <Loader2 className="h-4 w-4 animate-spin" />
   299→            ) : editing ? (
   300→              <PencilLine className="h-4 w-4" />
   301→            ) : (
   302→              <UserPlus className="h-4 w-4" />
   303→            )}
   304→            {editing ? "Simpan Perubahan" : "Buat"}
   305→          </Button>
   306→        </DialogFooter>
   307→      </DialogContent>
   308→    </Dialog>
   309→  );
   310→}

   312→/** Dialog edit pengumuman komunitas (pemilik/Super Admin). */
   313→function EditAnnouncementDialog({
   314→  community,
   315→  open,
   316→  onOpenChange,
   317→  onSaved,
   318→}: {
   319→  community: CommunityT;
   320→  open: boolean;
   321→  onOpenChange: (open: boolean) => void;
   322→  onSaved: (updated: CommunityT) => void;
   323→}) {
   324→  const { toast } = useToast();
   325→  const [text, setText] = useState(community.announcement);
   326→  const [saving, setSaving] = useState(false);
   327→
   328→  useEffect(() => {
   329→    if (open) setText(community.announcement);
   330→  }, [open, community.announcement]);
   331→
   332→  async function save() {
   333→    setSaving(true);
   334→    try {
   335→      const res = await fetch(`/api/communities/${community.id}`, {
   336→        method: "PATCH",
   337→        headers: { "Content-Type": "application/json" },
   338→        cache: "no-store",

   359→      <DialogContent className="sm:max-w-md">

   381→            Simpan Pengumuman

   389→/** Dialog kirim pengumuman ke satu grup komunitas — pengumuman NYATA via chat grup. */
   390→function SendGroupAnnouncementDialog({
   391→  community,
   392→  group,
   393→  open,
   394→  onOpenChange,
   395→  onAnnounced,
   396→}: {
   397→  community: CommunityT;
   398→  group: GroupT | null;
   399→  open: boolean;
   400→  onOpenChange: (open: boolean) => void;
   401→  onAnnounced: (updated: CommunityT) => void;
   402→}) {
   403→  const { toast } = useToast();
   404→  const [msg, setMsg] = useState("");
   405→  const [sending, setSending] = useState(false);
   406→
   407→  async function send() {
   408→    if (!group || sending) return;
   409→    setSending(true);
   410→    try {
   411→      const res = await fetch(`/api/communities/${community.id}/announce`, {
   412→        method: "POST",
   413→        headers: { "Content-Type": "application/json" },
   414→        body: JSON.stringify({ groupId: group.id, text: msg.trim() }),
   415→      });
   416→      const data = await res.json();
   417→      if (!res.ok || !data?.message) throw new Error(data?.error || "Gagal mengirim pengumuman");
   418→      if (data.community) onAnnounced(toCommunity(data.community as CommunityRaw));
   419→      toast({
   420→        title: "Pengumuman terkirim",
   421→        description: data.created
   422→          ? `Chat grup "${group.name}" dibuat otomatis dan pengumuman masuk ke chat tersebut.`
   423→          : `Pengumuman terkirim ke grup "${group.name}" — buka tab Chat untuk melihatnya.`,
   424→      });
   425→      setMsg("");
   426→      onOpenChange(false);
   427→    } catch (e) {
   428→      toast({
   429→        title: "Gagal mengirim pengumuman",
   430→        description: e instanceof Error ? e.message : "Silakan coba lagi.",
   431→        variant: "destructive",
   432→      });
   433→    } finally {
   434→      setSending(false);
   435→    }
   436→  }
   437→
   438→  return (
   439→    <Dialog
   440→      open={open}
   441→      onOpenChange={(v) => {
   442→        if (!v) setMsg("");
   443→        onOpenChange(v);
   444→      }}
   445→    >
   446→      <DialogContent className="sm:max-w-md">
   447→        <DialogHeader>
   448→          <DialogTitle className="flex items-center gap-2">
   449→            <Send className="h-5 w-5 text-primary" />
   450→            Kirim Pengumuman ke Grup
   451→          </DialogTitle>
   452→          <DialogDescription>
   453→            {group
   454→              ? `Grup "${group.name}" · ${community.name} — pengumuman dikirim sebagai pesan ke chat grup.`
   455→              : ""}
   456→          </DialogDescription>
   457→        </DialogHeader>
   458→        <Textarea
   459→          value={msg}
   460→          onChange={(e) => setMsg(e.target.value)}
   461→          rows={4}
   462→          maxLength={2000}
   463→          placeholder="Tulis pengumuman ke grup…"
   464→          aria-label="Isi pengumuman grup"
   465→        />
   466→        <DialogFooter>
   467→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={sending}>
   468→            Batal
   469→          </Button>
   470→          <Button onClick={() => void send()} disabled={sending || msg.trim().length === 0}>
   471→            {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   472→            Kirim
   473→          </Button>
   474→        </DialogFooter>
   475→      </DialogContent>
   476→    </Dialog>
   477→  );
   478→}

   579→/** Dialog konfirmasi hapus komunitas (Task 46). */
   580→function DeleteCommunityDialog({
   581→  community,
   582→  open,
   583→  onOpenChange,
   584→  onDeleted,
   585→}: {
   586→  community: CommunityT;
   587→  open: boolean;
   588→  onOpenChange: (open: boolean) => void;
   589→  onDeleted: (id: string) => void;
   590→}) {
   591→  const { toast } = useToast();
   592→  const [deleting, setDeleting] = useState(false);
   593→
   594→  async function remove() {
   595→    if (deleting) return;
   596→    setDeleting(true);
   597→    try {
   598→      const res = await fetch(`/api/communities/${community.id}`, { method: "DELETE", cache: "no-store" });
   599→      const data = await res.json();
   600→      if (!res.ok || !data?.ok) throw new Error(data?.error || "Gagal menghapus komunitas");
   601→      onDeleted(community.id);
   602→      toast({
   603→        title: "Komunitas dihapus",
   604→        description: `"${community.name}"${
   605→          data.deletedGroups ? ` beserta ${data.deletedGroups} grup di dalamnya` : ""
   606→        } telah dihapus.`,
   607→      });
   608→      onOpenChange(false);
   609→    } catch (e) {
   610→      toast({
   611→        title: "Gagal menghapus komunitas",
   612→        description: e instanceof Error ? e.message : "Silakan coba lagi.",
   613→        variant: "destructive",
   614→      });
   615→    } finally {
   616→      setDeleting(false);
   617→    }
   618→  }
   619→
   620→  return (
   621→    <AlertDialog open={open} onOpenChange={onOpenChange}>
   622→      <AlertDialogContent>
   623→        <AlertDialogHeader>
   624→          <AlertDialogTitle>Hapus komunitas?</AlertDialogTitle>
   625→          <AlertDialogDescription>
   626→            Komunitas <span className="font-semibold text-foreground">“{community.name}”</span> beserta{" "}
   627→            {community.groups.length} grup di dalamnya akan dihapus permanen. Chat grup yang sudah dibuat tidak ikut
   628→            terhapus. Tindakan ini tidak dapat dibatalkan.
   629→          </AlertDialogDescription>
   630→        </AlertDialogHeader>
   631→        <AlertDialogFooter>
   632→          <AlertDialogCancel disabled={deleting}>Batal</AlertDialogCancel>
   633→          <AlertDialogAction
   634→            className="bg-destructive text-white hover:bg-destructive/90 focus-visible:ring-destructive"
   635→            disabled={deleting}
   636→            onClick={(e) => {
   637→              e.preventDefault();
   638→              void remove();
   639→            }}
   640→          >
   641→            {deleting ? <Loader2 className="h-4 w-4 animate-spin" /> : <Trash2 className="h-4 w-4" />}
   642→            Hapus Komunitas
   643→          </AlertDialogAction>
   644→        </AlertDialogFooter>
   645→      </AlertDialogContent>
   646→    </AlertDialog>
   647→  );
   648→}
