   343→/**
   344→ * Toggle cepat tampilan (Mode Gelap & Mode Hemat) — dasar drawer mobile,
   345→ * sehingga bisa diakses dari semua view tanpa membuka Pengaturan.
   346→ */
   347→function DisplayToggles() {
   348→  const { theme, setTheme } = useTheme();
   349→  const lite = useDisplayPrefs((s) => s.lite);
   350→  const setLite = useDisplayPrefs((s) => s.setLite);
   351→  const isDark = theme === "dark";
   352→
   353→  return (
   354→    <div className="px-0 py-2 border-t border-border">
   355→      <p className="px-4 pt-2 pb-1 text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">Tampilan</p>
   356→      <button
   357→        type="button"
   358→        onClick={() => setTheme(isDark ? "light" : "dark")}
   359→        className="w-full flex items-center gap-3 px-4 py-2.5 text-sm text-left transition-colors hover:bg-secondary"
   360→        role="menuitemcheckbox"
   361→        aria-checked={isDark}
   362→      >
   363→        <Moon className={cn("h-[18px] w-[18px] shrink-0", isDark ? "text-primary" : "text-muted-foreground")} />
   364→        <span>Mode Gelap</span>
   365→        {isDark && <Check className="ml-auto h-4 w-4 text-primary" aria-label="Aktif" />}
   366→      </button>
   367→      <button
   368→        type="button"
   369→        onClick={() => setLite(!lite)}
   370→        className="w-full flex items-center gap-3 px-4 py-2.5 text-sm text-left transition-colors hover:bg-secondary"
   371→        role="menuitemcheckbox"
   372→        aria-checked={lite}
   373→      >
   374→        <Leaf className={cn("h-[18px] w-[18px] shrink-0", lite ? "text-primary" : "text-muted-foreground")} />
   375→        <span>Mode Hemat</span>
   376→        {lite && <Check className="ml-auto h-4 w-4 text-primary" aria-label="Aktif" />}
   377→      </button>
   378→    </div>
   379→  );
   380→}
