    66→function readLikes(): Set<string> {
    67→  try {
    68→    const raw = window.localStorage.getItem(LIKES_KEY);
    69→    return new Set<string>(raw ? (JSON.parse(raw) as string[]) : []);
    70→  } catch {
    71→    return new Set<string>();
    72→  }

    75→function timeAgo(iso: string): string {
    76→  const diff = Date.now() - new Date(iso).getTime();
    77→  const m = Math.floor(diff / 60000);
    78→  if (m < 1) return "baru saja";
    79→  if (m < 60) return `${m} mnt lalu`;
    80→  const h = Math.floor(m / 60);
    81→  if (h < 24) return `${h} jam lalu`;
    82→  const d = Math.floor(h / 24);
    83→  if (d < 7) return `${d} hari lalu`;
    84→  return new Date(iso).toLocaleDateString("id-ID", { day: "numeric", month: "short" });
    85→}

    87→export default function BerandaView({ onNavigate }: { onNavigate?: (view: ViewId) => void }) {
    88→  const { toast } = useToast();
    89→  const { user } = useAuth();

    91→  const [videos, setVideos] = useState<VideoPost[]>([]);
    92→  const [loading, setLoading] = useState(true);
    93→  const [muted, setMuted] = useState(true);
    94→  const [likedIds, setLikedIds] = useState<Set<string>>(new Set());
    95→  const [uploadOpen, setUploadOpen] = useState(false);
    96→  const [editTarget, setEditTarget] = useState<VideoPost | null>(null);
    97→  const [deleteTarget, setDeleteTarget] = useState<VideoPost | null>(null);
    98→  const feedRef = useRef<HTMLDivElement>(null);
    99→
   100→  useEffect(() => {
   101→    setLikedIds(readLikes());

   103→
   104→  const load = useCallback(async (t: TabId) => {
   105→    setLoading(true);
   106→    try {
   107→      const res = await fetch(`/api/videos?tab=${t}`, { cache: "no-store" });
   108→      const data = (await res.json()) as { videos?: VideoPost[] };
   109→      setVideos(data.videos ?? []);
   110→    } catch {
   111→      toast({ title: "Gagal memuat video", variant: "destructive" });

   113→      setLoading(false);
   114→    }
   115→  }, [toast]);
   116→
   117→  useEffect(() => {
   118→    void load(tab);
   119→  }, [tab, load]);
   120→
   121→  // ---------- Autoplay video yang terlihat ----------
   122→  useEffect(() => {
   123→    const root = feedRef.current;
   124→    if (!root) return;
   125→    const observer = new IntersectionObserver(
   126→      (entries) => {
   127→        entries.forEach((entry) => {
   128→          const vid = entry.target.querySelector("video");
   129→          if (!vid) return;
   130→          if (entry.isIntersecting && entry.intersectionRatio >= 0.6) {
   131→            vid.play().catch(() => {});
   132→          } else {
   133→            vid.pause();
   134→          }
   135→        });
   136→      },
   137→      { root, threshold: [0, 0.6] }
   138→    );
   139→    const items = root.querySelectorAll("[data-feed-item]");
   140→    items.forEach((el) => observer.observe(el));
   141→    return () => observer.disconnect();
   142→  }, [videos, loading]);
   143→
   144→  const toggleLike = async (v: VideoPost) => {
   145→    const liked = !likedIds.has(v.id);
   146→    const next = new Set(likedIds);
   147→    if (liked) next.add(v.id);
   148→    else next.delete(v.id);
   149→    setLikedIds(next);
   150→    try {
   151→      window.localStorage.setItem(LIKES_KEY, JSON.stringify(Array.from(next)));
   152→    } catch {
   153→      // abaikan
   154→    }
   155→    setVideos((prev) => prev.map((p) => (p.id === v.id ? { ...p, likes: Math.max(0, p.likes + (liked ? 1 : -1)) } : p)));
   156→    try {
   157→      await fetch(`/api/videos/${v.id}/like`, {
   158→        method: "POST",
   159→        headers: { "Content-Type": "application/json" },
   160→        body: JSON.stringify({ liked }),
   161→      });
   162→    } catch {
   163→      toast({ title: "Gagal menyimpan suka", variant: "destructive" });
   164→    }
   165→  };
   166→
   167→  const shareVideo = async (v: VideoPost) => {
   168→    const url = `${window.location.origin}${v.videoUrl}`;
   169→    try {
   170→      if (navigator.share) {
   171→        await navigator.share({ title: v.title, text: `Tonton "${v.title}" di Beranda`, url });
   172→        return;
   173→      }
   174→      throw new Error("no-share");
   175→    } catch {
   176→      try {
   177→        await navigator.clipboard.writeText(url);
   178→        toast({ title: "Tautan video disalin" });
   179→      } catch {
   180→        toast({ title: "Gagal menyalin tautan", variant: "destructive" });
   181→      }
   182→    }
   183→  };

   185→  // ---------- Kelola video (edit/hapus sesuai peran) ----------

   190→      if (!res.ok) throw new Error(data.error ?? "Gagal menghapus video");
   191→      setVideos((prev) => prev.filter((p) => p.id !== v.id));
   192→      toast({ title: "Video dihapus dari Beranda" });
   193→    } catch (e) {
   194→      toast({ title: e instanceof Error ? e.message : "Gagal menghapus video", variant: "destructive" });
   195→    } finally {
   196→      setDeleteTarget(null);
   197→    }

   308→// ---------- Satu item feed ----------
   309→function FeedItem({
   310→  video,
   311→  muted,
   312→  onToggleMute,
   313→  liked,
   314→  onLike,
   315→  onShare,
   316→  onAuthorClick,
   317→  canEdit,
   318→  onEdit,
   319→  onDelete,
   320→}: {
   321→  video: VideoPost;
   322→  muted: boolean;
   323→  onToggleMute: () => void;
   324→  liked: boolean;
   325→  onLike: () => void;
   326→  onShare: () => void;
   327→  onAuthorClick: () => void;
   328→  canEdit: boolean;
   329→  onEdit: () => void;
   330→  onDelete: () => void;
   331→}) {
   332→  const vidRef = useRef<HTMLVideoElement>(null);
   333→  const [paused, setPaused] = useState(false);
   334→  const [progress, setProgress] = useState(0);
   335→
   336→  const togglePlay = () => {
   337→    const vid = vidRef.current;
   338→    if (!vid) return;
   339→    if (vid.paused) {
   340→      vid.play().catch(() => {});
   341→      setPaused(false);
   342→    } else {
   343→      vid.pause();
   344→      setPaused(true);
   345→    }
   346→  };
   347→
   348→  const initials = video.authorName.slice(0, 2).toUpperCase();
   349→
   350→  return (
   351→    <div data-feed-item className="relative h-full w-full snap-start snap-always overflow-hidden bg-neutral-950">
   352→      <video
   353→        ref={vidRef}
   354→        src={video.videoUrl}
   355→        poster={video.posterUrl || undefined}
   356→        loop
   357→        muted={muted}
   358→        playsInline
   359→        preload="metadata"
   360→        className="absolute inset-0 h-full w-full object-cover"
   361→        onClick={togglePlay}
   362→        onTimeUpdate={(e) => {
   363→          const el = e.currentTarget;
   364→          setProgress(el.duration ? (el.currentTime / el.duration) * 100 : 0);
   365→        }}
   366→        aria-label={`Video: ${video.title}`}
   367→      />
   368→
   369→      {/* Tombol play saat dijeda */}
   370→      {paused && (
   371→        <button
   372→          type="button"
   373→          onClick={togglePlay}
   374→          aria-label="Putar video"
   375→          className="absolute inset-0 flex items-center justify-center"
   376→        >
   377→          <span className="h-16 w-16 rounded-full bg-black/50 flex items-center justify-center backdrop-blur-sm">
   378→            <Play className="h-7 w-7 text-white fill-white" />
   379→          </span>
   380→        </button>
   381→      )}
   382→
   383→      {/* Toggle suara */}
   384→      <button
   385→        type="button"
   386→        onClick={onToggleMute}
   387→        aria-label={muted ? "Nyalakan suara" : "Bisukan suara"}
   388→        className="absolute top-3 right-3 h-9 w-9 rounded-full bg-black/45 text-white flex items-center justify-center backdrop-blur-sm hover:bg-black/60 transition-colors z-10"
   389→      >
   390→        {muted ? <VolumeX className="h-4 w-4" /> : <Volume2 className="h-4 w-4" />}
   391→      </button>
   392→
   393→      {/* Aksi kanan */}
   394→      <div className="absolute right-3 bottom-28 flex flex-col items-center gap-4 z-10">
   395→        <button
   396→          type="button"
   397→          onClick={onLike}
   398→          aria-label={liked ? "Batal suka" : "Suka video"}
   399→          aria-pressed={liked}
   400→          className="flex flex-col items-center gap-1 text-white"
   401→        >
   402→          <span className="h-11 w-11 rounded-full bg-black/45 flex items-center justify-center backdrop-blur-sm hover:bg-black/60 transition-colors">
   403→            <Heart className={cn("h-6 w-6", liked && "fill-red-500 text-red-500")} />
   404→          </span>
   405→          <span className="text-[11px] font-semibold">{video.likes.toLocaleString("id-ID")}</span>
   406→        </button>
   407→        <button
   408→          type="button"
   409→          onClick={onShare}
   410→          aria-label="Bagikan video"
   411→          className="flex flex-col items-center gap-1 text-white"
   412→        >
   413→          <span className="h-11 w-11 rounded-full bg-black/45 flex items-center justify-center backdrop-blur-sm hover:bg-black/60 transition-colors">
   414→            <Share2 className="h-5 w-5" />
   415→          </span>
   416→          <span className="text-[11px] font-semibold">Bagikan</span>
   417→        </button>
   418→        {canEdit && (
   419→          <>
   420→            <button
   421→              type="button"
   422→              onClick={onEdit}
   423→              aria-label="Edit video"
   424→              className="flex flex-col items-center gap-1 text-white"
   425→            >
   426→              <span className="h-11 w-11 rounded-full bg-black/45 flex items-center justify-center backdrop-blur-sm hover:bg-black/60 transition-colors">
   427→                <Pencil className="h-5 w-5" />
   428→              </span>
   429→              <span className="text-[11px] font-semibold">Edit</span>
   430→            </button>
   431→            <button
   432→              type="button"
   433→              onClick={onDelete}
   434→              aria-label="Hapus video"
   435→              className="flex flex-col items-center gap-1 text-white"
   436→            >
   437→              <span className="h-11 w-11 rounded-full bg-black/45 flex items-center justify-center backdrop-blur-sm hover:bg-black/60 transition-colors">
   438→                <Trash2 className="h-5 w-5 text-red-400" />
   439→              </span>
   440→              <span className="text-[11px] font-semibold">Hapus</span>
   441→            </button>
   442→          </>
   443→        )}
   444→      </div>
   445→
   446→      {/* Info penulis */}
   447→      <div className="absolute left-0 right-16 bottom-0 p-4 pb-6 z-10 text-white">
   448→        <button
   449→          type="button"
   450→          onClick={onAuthorClick}
   451→          className="flex items-center gap-2.5 mb-2 group"
   452→          aria-label={`Lihat penjual ${video.authorName}`}

   454→          <span className="h-9 w-9 rounded-full bg-gradient-to-br from-emerald-400 to-teal-600 flex items-center justify-center text-[13px] font-bold ring-2 ring-white/60 shrink-0">

   459→          {video.isFollower && <Users className="h-3.5 w-3.5 text-sky-300" aria-label="Pengikut" />}
   460→        </button>
   461→        <p className="text-sm leading-snug drop-shadow line-clamp-2">{video.title}</p>

   464→            {CATEGORY_LABEL[video.category] ?? video.category}

   468→          </span>
   469→          <span className="text-[11px] text-white/50">{timeAgo(video.createdAt)}</span>
   470→        </div>
   471→      </div>
   472→
   473→      {/* Progres */}
   474→      <div className="absolute bottom-0 left-0 right-0 h-[3px] bg-white/15 z-10">
   475→        <div className="h-full bg-white/80 transition-[width] duration-150" style={{ width: `${progress}%` }} />
   476→      </div>
   477→    </div>
   478→  );
   479→}

   481→// ---------- Dialog kirim video ----------
   482→function UploadDialog({
   483→  open,
   484→  onOpenChange,
   485→  onUploaded,
   486→}: {
   487→  open: boolean;
   488→  onOpenChange: (v: boolean) => void;
   489→  onUploaded: (v: VideoPost) => void;
   490→}) {
   491→  const { toast } = useToast();
   492→  const [file, setFile] = useState<File | null>(null);
   493→  const [title, setTitle] = useState("");
   494→  const [category, setCategory] = useState<TabId>("hiburan");
   495→  const [busy, setBusy] = useState(false);
   496→  const [previewUrl, setPreviewUrl] = useState<string>("");
   497→
   498→  const reset = () => {
   499→    setFile(null);
   500→    setTitle("");
   501→    setCategory("hiburan");
   502→    setPreviewUrl("");
   503→  };
   504→
   505→  const submit = async () => {
   506→    if (!file || !title.trim()) return;
   507→    setBusy(true);
   508→    try {
   509→      const fd = new FormData();
   510→      fd.append("file", file);
   511→      fd.append("kind", "video");
   512→      const up = await fetch("/api/upload", { method: "POST", body: fd });
   513→      const upData = (await up.json()) as { url?: string; error?: string };
   514→      if (!up.ok || !upData.url) throw new Error(upData.error ?? "Gagal mengunggah video");
   515→
   516→      const res = await fetch("/api/videos", {
   517→        method: "POST",
   518→        headers: { "Content-Type": "application/json" },
   519→        body: JSON.stringify({ title: title.trim(), category, videoUrl: upData.url, authorName: "Saya" }),
   520→      });
   521→      const data = (await res.json()) as { video?: VideoPost; error?: string };
   522→      if (!res.ok || !data.video) throw new Error(data.error ?? "Gagal mengirim video");
   523→
   524→      toast({ title: "Video terkirim ke Beranda 🎉" });
   525→      onUploaded(data.video);
   526→      onOpenChange(false);
   527→      reset();
   528→    } catch (e) {
   529→      toast({ title: e instanceof Error ? e.message : "Gagal mengirim video", variant: "destructive" });
   530→    } finally {
   531→      setBusy(false);
   532→    }
   533→  };
   534→
   535→  return (
   536→    <Dialog open={open} onOpenChange={(v) => { if (!busy) onOpenChange(v); }}>
   537→      <DialogContent className="sm:max-w-md">
   538→        <DialogHeader>
   539→          <DialogTitle>Kirim Video ke Beranda</DialogTitle>

   560→                htmlFor="video-file"

   575→                setFile(f);

   577→              }}

   579→          </div>
   580→          <div className="space-y-1.5">
   581→            <Label htmlFor="video-title">Judul</Label>

   583→              id="video-title"

   593→              {(["hiburan", "domisili", "promo"] as const).map((c) => (
   594→                <button

   596→                  type="button"

   599→                  onClick={() => setCategory(c)}

   601→                    "px-3 py-1.5 rounded-full text-[13px] transition-colors",
   602→                    category === c ? "bg-primary text-primary-foreground font-semibold" : "bg-secondary text-muted-foreground hover:bg-secondary/70"

   604→                >
   605→                  {CATEGORY_LABEL[c]}
   606→                </button>
   607→              ))}
   608→            </div>
   609→          </div>
   610→        </div>
   611→        <DialogFooter>
   612→          <Button variant="ghost" onClick={() => onOpenChange(false)} disabled={busy}>

   616→            {busy && <Loader2 className="h-4 w-4 animate-spin" />} Kirim
   617→          </Button>
   618→        </DialogFooter>

   624→// ---------- Dialog edit video ----------
   625→function VideoEditDialog({
   626→  open,
   627→  onOpenChange,
   628→  video,
   629→  onSaved,
   630→}: {
   631→  open: boolean;
   632→  onOpenChange: (v: boolean) => void;
   633→  video: VideoPost | null;
   634→  onSaved: (v: VideoPost) => void;
   635→}) {
   636→  const { toast } = useToast();
   637→  const [title, setTitle] = useState("");
   638→  const [category, setCategory] = useState<TabId>("hiburan");
   639→  const [busy, setBusy] = useState(false);
   640→
   641→  useEffect(() => {
   642→    if (!open || !video) return;
   643→    setTitle(video.title);
   644→    setCategory((video.category as TabId) ?? "hiburan");
   645→  }, [open, video]);
   646→
   647→  const submit = async () => {
   648→    if (!video || !title.trim()) return;
   649→    setBusy(true);
   650→    try {
   651→      const res = await fetch(`/api/videos/${video.id}`, {
   652→        method: "PATCH",
   653→        headers: { "Content-Type": "application/json" },
   654→        body: JSON.stringify({ title: title.trim(), category }),
   655→      });
   656→      const data = (await res.json()) as { video?: VideoPost; error?: string };
   657→      if (!res.ok || !data.video) throw new Error(data.error ?? "Gagal menyimpan perubahan");
   658→      onSaved(data.video);
   659→      onOpenChange(false);
   660→    } catch (e) {
   661→      toast({ title: e instanceof Error ? e.message : "Gagal menyimpan perubahan", variant: "destructive" });
   662→    } finally {
   663→      setBusy(false);
   664→    }
   665→  };
   666→
   667→  return (
   668→    <Dialog open={open} onOpenChange={(v) => { if (!busy) onOpenChange(v); }}>
   669→      <DialogContent className="sm:max-w-md">
   670→        <DialogHeader>
   671→          <DialogTitle>Edit Video</DialogTitle>
   672→          <DialogDescription>Ubah judul atau kategori video di feed Beranda.</DialogDescription>
   673→        </DialogHeader>
   674→        <div className="space-y-4">
   675→          <div className="space-y-1.5">
   676→            <Label htmlFor="edit-video-title">Judul</Label>
   677→            <Textarea
   678→              id="edit-video-title"
   679→              value={title}
   680→              onChange={(e) => setTitle(e.target.value)}
   681→              placeholder="Judul video"
   682→              rows={2}
   683→              maxLength={120}
   684→            />
   685→          </div>
   686→          <div className="space-y-1.5">
   687→            <Label>Kategori</Label>
   688→            <div className="flex gap-1.5" role="radiogroup" aria-label="Kategori video">
   689→              {(["hiburan", "domisili", "promo"] as const).map((c) => (
   690→                <button
   691→                  key={c}
   692→                  type="button"
   693→                  role="radio"
   694→                  aria-checked={category === c}
   695→                  onClick={() => setCategory(c)}
   696→                  className={cn(
   697→                    "px-3 py-1.5 rounded-full text-[13px] transition-colors",
   698→                    category === c ? "bg-primary text-primary-foreground font-semibold" : "bg-secondary text-muted-foreground hover:bg-secondary/70"
   699→                  )}
   700→                >
   701→                  {CATEGORY_LABEL[c]}
   702→                </button>
   703→              ))}
   704→            </div>
   705→          </div>
   706→        </div>
   707→        <DialogFooter>
   708→          <Button variant="ghost" onClick={() => onOpenChange(false)} disabled={busy}>
   709→            Batal
   710→          </Button>
   711→          <Button onClick={() => void submit()} disabled={busy || !title.trim()}>
   712→            {busy && <Loader2 className="h-4 w-4 animate-spin" />} Simpan
   713→          </Button>
   714→        </DialogFooter>
   715→      </DialogContent>
   716→    </Dialog>
   717→  );
   718→}
