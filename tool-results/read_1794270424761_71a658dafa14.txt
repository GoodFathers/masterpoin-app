     1→     1→"use client";
     2→     2→// ===================== MESIN PANGGILAN NYATA INDIIGOO =====================
     3→     3→// Client-side singleton untuk panggilan suara & video NYATA antar pengguna
     4→     4→// IndiGoo (WebRTC) + keanggotaan realtime via signaling socket.io
     5→     5→// (mini-services/call-service, diakses lewat "/?XTransformPort=3003").
     6→     6→//
     7→     7→// Tanggung jawab:
     8→     8→//  - Koneksi socket + auth token sesi + presence (siapa yang daring).
     9→     9→//  - Siklus hidup panggilan 1-on-1: invite → ring → accept/reject/missed →
    10→    10→//    offer/answer/ICE → aktif (timer) → end, termasuk fallback audio-only
    11→    11→//    bila kamera tidak tersedia, dan pembersihan resource yang aman.
    12→    12→//  - Menyimpan riwayat panggilan ke /api/calls (OUT utk penelepon, IN utk
    13→    13→//    penerima, MISSED bila tak dijawab).
    14→    14→//  - Konferensi mesh kecil: join/leave ruangan + relay offer/answer/ICE.
    15→    15→//
    16→    16→// Semua state tampilan hidup di store Zustand (src/store/realcall.ts) —
    17→    17→// mesin ini yang menuliskannya. UI (RealCallCenter) hanya membaca & memanggil
    18→    18→// aksi mesin. Server signaling TIDAK pernah menyentuh media.
    19→    19→
    20→    20→import type { Socket } from "socket.io-client";
    21→    21→import {
    22→    22→  playCallEndTone,
    23→    23→  startIncomingRing,
    24→    24→  startRingback,
    25→    25→  stopIncomingRing,
    26→    26→  stopRingback,
    27→    27→} from "@/lib/alarm";
    28→    28→import { getSessionToken } from "@/lib/session-token";
    29→    29→import { useRealcallStore, type CallKind, type PeerInfo, type CallPhase } from "@/store/realcall";
    30→    30→
    31→    31→const SIGNAL_URL = "/?XTransformPort=3003";
    32→    32→const ICE_SERVERS: RTCIceServer[] = [
    33→    33→  { urls: ["stun:stun.l.google.com:19302", "stun:stun1.l.google.com:19302"] },
    34→    34→];
    35→    35→
    36→    36→export interface ActiveCall {
    37→    37→  callId: string;
    38→    38→  role: "caller" | "callee";
    39→    39→  peer: PeerInfo;
    40→    40→  kind: CallKind;
    41→    41→  phase: CallPhase; // ringing | connecting | active
    42→    42→  startedAt: number | null; // aktif sejak (epoch ms)
    43→    43→  videoFallback: boolean; // video diminta tapi kamera tak tersedia → audio-only
    44→    44→  muted: boolean;
    45→    45→  camOff: boolean;
    46→    46→}
    47→    47→
    48→    48→class RealCallEngine {
    49→    49→  private socket: Socket | null = null;
    50→    50→  private connecting = false;
    51→    51→  private generation = 0; // penanda siklus connect/disconnect (anti race StrictMode)
    52→    52→
    53→    53→  // WebRTC (1-on-1)
    54→    54→  private pc: RTCPeerConnection | null = null;
    55→    55→  private localStream: MediaStream | null = null;
    56→    56→  private remoteStream: MediaStream | null = null;
    57→    57→  private call: ActiveCall | null = null;
    58→    58→  private queuedCandidates: RTCIceCandidateInit[] = [];
    59→    59→  private makingOffer = false;
    60→    60→
    61→    61→  // Konferensi mesh
    62→    62→  private confRoomId: string | null = null;
    63→    63→  private confPeers = new Map<string, RTCPeerConnection>(); // peerSocketId -> pc
    64→    64→  private confStreams = new Map<string, MediaStream>(); // peerSocketId -> remote
    65→    65→  private confLocalStream: MediaStream | null = null;
    66→    66→
    67→    67→  /** Fallback media sintetis untuk lingkungan tanpa kamera/mik (sandbox/E2E). */
    68→    68→  static syntheticRequested = false;
    69→    69→
    70→    70→  // ---------------- KONEKSI & PRESENCE ----------------
    71→    71→
    72→    72→  ensureConnected(): void {
    73→    73→    if (typeof window === "undefined" || this.socket || this.connecting) return;
    74→    74→    const token = getSessionToken();
    75→    75→    if (!token) return; // belum login — jangan buka koneksi
    76→    76→    this.connecting = true;
    77→    77→    const gen = this.generation;
    78→    78→
    79→    79→    void import("socket.io-client").then(({ io }) => {
    80→    80→      if (gen !== this.generation) return; // sudah di-disconnect saat import berjalan
    81→    81→      const s = io(SIGNAL_URL, {
    82→    82→        path: "/", // DO NOT change — dipakai Caddy
    83→    83→        transports: ["websocket", "polling"],
    84→    84→        reconnection: true,
    85→    85→        reconnectionAttempts: 8,
    86→    86→        reconnectionDelay: 1200,
    87→    87→        timeout: 10000,
    88→    88→      });
    89→    89→      this.socket = s;
    90→    90→
    91→    91→      s.on("connect", () => {
    92→    92→        s.emit("auth", { token }, (res: { ok: boolean; user?: PeerInfo; peers?: PeerInfo[]; error?: string }) => {
    93→    93→          this.connecting = false;
    94→    94→          const st = useRealcallStore.getState();
    95→    95→          if (!res?.ok || !res.user) {
    96→    96→            st.setConn("error");
    97→    97→            return;
    98→    98→          }
    99→    99→          st.setMe(res.user);
   100→   100→          st.setPeers(res.peers ?? []);
   101→   101→          st.setConn("connected");
   102→   102→        });
   103→   103→      });
   104→   104→
   105→   105→      s.on("disconnect", () => {
   106→   106→        useRealcallStore.getState().setConn("connecting");
   107→   107→      });
   108→   108→
   109→   109→      s.on("presence:update", (peers: PeerInfo[]) => {
   110→   110→        useRealcallStore.getState().setPeers(peers ?? []);
   111→   111→      });
   112→   112→
   113→   113→      // ---------------- panggilan masuk ----------------
   114→   114→      s.on("call:invite", (p: { callId: string; from: PeerInfo; kind: CallKind }) => {
   115→   115→        const st = useRealcallStore.getState();
   116→   116→        if (this.call || st.incoming) {
   117→   117→          // sibuk — tolak otomatis
   118→   118→          s.emit("call:reject", { callId: p.callId });
   119→   119→          return;
   120→   120→        }
   121→   121→        st.setIncoming({ callId: p.callId, from: p.from, kind: p.kind, at: Date.now() });
   122→   122→        startIncomingRing();
   123→   123→      });
   124→   124→
   125→   125→      s.on("call:cancelled", ({ callId }: { callId: string }) => {
   126→   126→        const st = useRealcallStore.getState();
   127→   127→        if (st.incoming?.callId === callId) {
   128→   128→          st.setIncoming(null);
   129→   129→          stopIncomingRing();
   130→   130→          playCallEndTone("end");
   131→   131→        }
   132→   132→      });
   133→   133→
   134→   134→      s.on("call:missed", (p: { callId: string; reason: string }) => {
   135→   135→        // Dipanggil di KEDUA sisi saat timeout
   136→   136→        const st = useRealcallStore.getState();
   137→   137→        const wasCaller = this.call?.role === "caller" && this.call.callId === p.callId;
   138→   138→        st.setIncoming(null);
   139→   139→        stopIncomingRing();
   140→   140→        stopRingback();
   141→   141→        playCallEndTone("busy");
   142→   142→        if (this.call && this.call.callId === p.callId) {
   143→   143→          this.saveHistory(wasCaller ? "OUT" : "MISSED", 0);
   144→   144→          this.cleanupCall();
   145→   145→        }
   146→   146→      });
   147→   147→
   148→   148→      s.on("call:rejected", ({ callId }: { callId: string }) => {
   149→   149→        if (this.call?.callId === callId && this.call.role === "caller") {
   150→   150→          stopRingback();
   151→   151→          playCallEndTone("busy");
   152→   152→          this.saveHistory("OUT", 0); // panggilan keluar, tidak tersambung
   153→   153→          this.cleanupCall();
   154→   154→        }
   155→   155→      });
   156→   156→
   157→   157→      s.on("call:accepted", ({ callId }: { callId: string }) => {
   158→   158→        if (this.call?.callId === callId && this.call.role === "caller") {
   159→   159→          stopRingback();
   160→   160→          void this.startMediaAndOffer();
   161→   161→        }
   162→   162→      });
   163→   163→
   164→   164→      s.on("call:offer", ({ callId, sdp }: { callId: string; sdp: RTCSessionDescriptionInit }) => {
   165→   165→        void this.handleOffer(callId, sdp);
   166→   166→      });
   167→   167→
   168→   168→      s.on("call:answer", ({ callId, sdp }: { callId: string; sdp: RTCSessionDescriptionInit }) => {
   169→   169→        void this.handleAnswer(callId, sdp);
   170→   170→      });
   171→   171→
   172→   172→      s.on("call:ice", ({ callId, candidate }: { callId: string; candidate: RTCIceCandidateInit }) => {
   173→   173→        void this.handleCandidate(callId, candidate);
   174→   174→      });
   175→   175→
   176→   176→      s.on("call:ended", ({ callId, reason }: { callId: string; reason?: string }) => {
   177→   177→        if (this.call?.callId === callId) {
   178→   178→          const dur = this.call.phase === "active" && this.call.startedAt ? Math.round((Date.now() - this.call.startedAt) / 1000) : 0;
   179→   179→          stopRingback();
   180→   180→          playCallEndTone("end");
   181→   181→          if (this.call.role === "callee" && dur === 0 && reason !== "hangup") {
   182→   182→            this.saveHistory("MISSED", 0);
   183→   183→          } else {
   184→   184→            this.saveHistory(this.call.role === "caller" ? "OUT" : "IN", dur);
   185→   185→          }
   186→   186→          this.cleanupCall();
   187→   187→        }
   188→   188→      });
   189→   189→
   190→   190→      // ---------------- konferensi ----------------
   191→   191→      s.on("conf:peer-joined", (p: { roomId: string; peerSocketId: string; userId: string; username: string; accessId: string }) => {
   192→   192→        void this.confCreateOfferTo(p.peerSocketId, { userId: p.userId, username: p.username, accessId: p.accessId });
   193→   193→      });
   194→   194→
   195→   195→      s.on("conf:peer-left", ({ peerSocketId }: { peerSocketId: string }) => {
   196→   196→        this.confClosePeer(peerSocketId);
   197→   197→        useRealcallStore.getState().removeConfStream(peerSocketId);
   198→   198→      });
   199→   199→
   200→   200→      s.on("conf:relay", (p: { fromPeer: string; fromUser?: { userId: string; username: string }; kind: "offer" | "answer" | "ice"; payload: unknown }) => {
   201→   201→        void this.confHandleRelay(p);
   202→   202→      });
   203→   203→
   204→   204→      s.on("conf:invite", (p: { roomId: string; from: PeerInfo }) => {
   205→   205→        // Abaikan bila sedang sibuk (panggilan 1-on-1 atau sudah di konferensi)
   206→   206→        const st = useRealcallStore.getState();
   207→   207→        if (this.call || st.confActive) return;
   208→   208→        st.setConfInvite({ roomId: p.roomId, from: p.from });
   209→   209→      });
   210→   210→
   211→   211→      s.io.on("reconnect", () => {
   212→   212→        // reconnect → auth ulang
   213→   213→        s.emit("auth", { token: getSessionToken() }, () => undefined);
   214→   214→      });
   215→   215→    });
   216→   216→  }
   217→   217→
   218→   218→  disconnect(): void {
   219→   219→    this.generation++;
   220→   220→    this.connecting = false;
   221→   221→    this.end("logout");
   222→   222→    this.leaveConf();
   223→   223→    this.socket?.disconnect();
   224→   224→    this.socket = null;
   225→   225→    const st = useRealcallStore.getState();
   226→   226→    st.reset();
   227→   227→  }
   228→   228→
   229→   229→  private emit(event: string, payload?: unknown): boolean {
   230→   230→    if (!this.socket?.connected) return false;
   231→   231→    this.socket.emit(event, payload);
   232→   232→    return true;
   233→   233→  }
   234→   234→
   235→   235→  // ---------------- PANGGILAN 1-ON-1 ----------------
   236→   236→
   237→   237→  /** Mulai panggilan keluar (penelepon). */
   238→   238→  invite(peer: PeerInfo, kind: CallKind): { ok: boolean; error?: string } {
   239→   239→    if (this.call) return { ok: false, error: "Ada panggilan yang sedang berjalan" };
   240→   240→    const st = useRealcallStore.getState();
   241→   241→    if (st.incoming) return { ok: false, error: "Ada panggilan masuk yang belum direspon" };
   242→   242→    const callId = `c-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 9)}`;
   243→   243→    this.call = {
   244→   244→      callId,
   245→   245→      role: "caller",
   246→   246→      peer,
   247→   247→      kind,
   248→   248→      phase: "ringing",
   249→   249→      startedAt: null,
   250→   250→      videoFallback: false,
   251→   251→      muted: false,
   252→   252→      camOff: false,
   253→   253→    };
   254→   254→    st.setCall(this.call);
   255→   255→    startRingback();
   256→   256→    const sent = this.emit("call:invite", { callId, to: peer.userId, kind });
   257→   257→    if (!sent) {
   258→   258→      stopRingback();
   259→   259→      this.cleanupCall();
   260→   260→      return { ok: false, error: "Tidak terhubung ke server realtime" };
   261→   261→    }
   262→   262→    return { ok: true };
   263→   263→  }
   264→   264→
   265→   265→  /** Penerima menekan terima. */
   266→   266→  accept(): void {
   267→   267→    const st = useRealcallStore.getState();
   268→   268→    const inc = st.incoming;
   269→   269→    if (!inc) return;
   270→   270→    stopIncomingRing();
   271→   271→    this.call = {
   272→   272→      callId: inc.callId,
   273→   273→      role: "callee",
   274→   274→      peer: inc.from,
   275→   275→      kind: inc.kind,
   276→   276→      phase: "connecting",
   277→   277→      startedAt: null,
   278→   278→      videoFallback: false,
   279→   279→      muted: false,
   280→   280→      camOff: false,
   281→   281→    };
   282→   282→    st.setIncoming(null);
   283→   283→    st.setCall(this.call);
   284→   284→    this.emit("call:accept", { callId: inc.callId });
   285→   285→    // Media + answer dibuat saat offer dari caller tiba (handleOffer)
   286→   286→  }
   287→   287→
   288→   288→  /** Penerima menolak. */
   289→   289→  reject(): void {
   290→   290→    const st = useRealcallStore.getState();
   291→   291→    const inc = st.incoming;
   292→   292→    if (!inc) return;
   293→   293→    stopIncomingRing();
   294→   294→    st.setIncoming(null);
   295→   295→    this.emit("call:reject", { callId: inc.callId });
   296→   296→    playCallEndTone("end");
   297→   297→  }
   298→   298→
   299→   299→  /** Akhiri panggilan dari sisi mana pun. */
   300→   300→  end(reason: string = "hangup"): void {
   301→   301→    if (!this.call) {
   302→   302→      // masih dering masuk tapi belum accept → anggap tolak
   303→   303→      this.reject();
   304→   304→      return;
   305→   305→    }
   306→   306→    const call = this.call;
   307→   307→    const dur = call.phase === "active" && call.startedAt ? Math.round((Date.now() - call.startedAt) / 1000) : 0;
   308→   308→    this.emit("call:end", { callId: call.callId, reason });
   309→   309→    stopRingback();
   310→   310→    stopIncomingRing();
   311→   311→    playCallEndTone("end");
   312→   312→    this.saveHistory(call.role === "caller" ? "OUT" : dur > 0 ? "IN" : "MISSED", dur);
   313→   313→    this.cleanupCall();
   314→   314→  }
   315→   315→
   316→   316→  toggleMute(): void {
   317→   317→    if (!this.call || !this.localStream) return;
   318→   318→    const next = !this.call.muted;
   319→   319→    for (const t of this.localStream.getAudioTracks()) t.enabled = !next;
   320→   320→    this.call = { ...this.call, muted: next };
   321→   321→    useRealcallStore.getState().setCall(this.call);
   322→   322→  }
   323→   323→
   324→   324→  async toggleCamera(): Promise<void> {
   325→   325→    if (!this.call || this.call.kind !== "VIDEO" || !this.localStream) return;
   326→   326→    const next = !this.call.camOff;
   327→   327→    try {
   328→   328→      if (next) {
   329→   329→        for (const t of this.localStream.getVideoTracks()) t.enabled = false;
   330→   330→      } else {
   331→   331→        for (const t of this.localStream.getVideoTracks()) t.enabled = true;
   332→   332→      }
   333→   333→      this.call = { ...this.call, camOff: next };
   334→   334→      useRealcallStore.getState().setCall(this.call);
   335→   335→    } catch {
   336→   336→      // abaikan
   337→   337→    }
   338→   338→  }

   340→   340→  /** (Penelepon) setelah di-accept: ambil media, buat PC, kirim offer. */
   341→   341→  private async startMediaAndOffer(): Promise<void> {
   342→   342→    if (!this.call || this.call.role !== "caller") return;
   343→   343→    this.setPhase("connecting");
   344→   344→    const ok = await this.setupLocalMedia(this.call.kind);
   345→   345→    if (!ok || !this.call) return;
   346→   346→    await this.createPeer();
   347→   347→    if (!this.pc || !this.localStream) return;
   348→   348→    for (const t of this.localStream.getTracks()) this.pc.addTrack(t, this.localStream);
   349→   349→    try {
   350→   350→      this.makingOffer = true;
   351→   351→      const offer = await this.pc.createOffer();
   352→   352→      await this.pc.setLocalDescription(offer);
   353→   353→      this.emit("call:offer", { callId: this.call.callId, sdp: this.pc.localDescription });
   354→   354→    } catch {
   355→   355→      this.failCall("Gagal menyiapkan koneksi");
   356→   356→    } finally {
   357→   357→      this.makingOffer = false;
   358→   358→    }
   359→   359→  }
   360→   360→
   361→   361→  /** (Penerima) offer tiba → ambil media, jawab. */
   362→   362→  private async handleOffer(callId: string, sdp: RTCSessionDescriptionInit): Promise<void> {
   363→   363→    if (!this.call || this.call.callId !== callId || this.call.role !== "callee") return;
   364→   364→    this.setPhase("connecting");
   365→   365→    const ok = await this.setupLocalMedia(this.call.kind);
   366→   366→    if (!ok || !this.call) return;
   367→   367→    await this.createPeer();
   368→   368→    if (!this.pc) return;
   369→   369→    try {
   370→   370→      await this.pc.setRemoteDescription(new RTCSessionDescription(sdp));
   371→   371→      for (const t of this.localStream?.getTracks() ?? []) this.pc.addTrack(t, this.localStream!);
   372→   372→      const answer = await this.pc.createAnswer();
   373→   373→      await this.pc.setLocalDescription(answer);
   374→   374→      this.emit("call:answer", { callId, sdp: this.pc.localDescription });
   375→   375→      for (const c of this.queuedCandidates) await this.pc.addIceCandidate(new RTCIceCandidate(c)).catch(() => undefined);
   376→   376→      this.queuedCandidates = [];
   377→   377→    } catch {
   378→   378→      this.failCall("Gagal menjawab panggilan");
   379→   379→    }
   380→   380→  }
   381→   381→
   382→   382→  private async handleAnswer(callId: string, sdp: RTCSessionDescriptionInit): Promise<void> {
   383→   383→    if (!this.pc || !this.call || this.call.callId !== callId) return;
   384→   384→    try {
   385→   385→      await this.pc.setRemoteDescription(new RTCSessionDescription(sdp));
   386→   386→      for (const c of this.queuedCandidates) await this.pc.addIceCandidate(new RTCIceCandidate(c)).catch(() => undefined);
   387→   387→      this.queuedCandidates = [];
   388→   388→    } catch {
   389→   389→      this.failCall("Negosiasi koneksi gagal");
   390→   390→    }
   391→   391→  }
   392→   392→
   393→   393→  private async handleCandidate(callId: string, candidate: RTCIceCandidateInit): Promise<void> {
   394→   394→    if (!this.call || this.call.callId !== callId) return;
   395→   395→    if (!this.pc || !this.pc.remoteDescription) {
   396→   396→      this.queuedCandidates.push(candidate);
   397→   397→      return;
   398→   398→    }
   399→   399→    await this.pc.addIceCandidate(new RTCIceCandidate(candidate)).catch(() => undefined);
   400→   400→  }

   402→   402→  /** Ambil mic (+ kamera utk VIDEO). Gagal kamera → audio-only (videoFallback). */
   403→   403→  private async setupLocalMedia(kind: CallKind): Promise<boolean> {
   404→   404→    if (this.localStream) return true;
   405→   405→    const wantVideo = kind === "VIDEO";
   406→   406→    try {
   407→   407→      this.localStream = await navigator.mediaDevices.getUserMedia({
   408→   408→        audio: true,
   409→   409→        video: wantVideo ? { width: { ideal: 1280 }, height: { ideal: 720 }, facingMode: "user" } : false,
   410→   410→      });
   411→   411→      useRealcallStore.getState().setLocalStream(this.localStream);
   412→   412→      return true;
   413→   413→    } catch {
   414→   414→      if (wantVideo) {
   415→   415→        try {
   416→   416→          this.localStream = await navigator.mediaDevices.getUserMedia({ audio: true });
   417→   417→          if (this.call) {
   418→   418→            this.call = { ...this.call, videoFallback: true, kind: "VOICE" };
   419→   419→            useRealcallStore.getState().setCall(this.call);
   420→   420→          }
   421→   421→          useRealcallStore.getState().setLocalStream(this.localStream);
   422→   422→          return true;
   423→   423→        } catch {
   424→   424→          this.failCall("Mikrofon & kamera tidak dapat diakses — izinkan akses di browser.");
   425→   425→          return false;
   426→   426→        }
   427→   427→      }
   428→   428→      this.failCall("Mikrofon tidak dapat diakses — izinkan akses mikrofon di browser.");
   429→   429→      return false;
   430→   430→    }

   433→   433→  private createPeer(): Promise<void> {
   434→   434→    return new Promise((resolve) => {
   435→   435→      if (this.pc) return resolve();
   436→   436→      const pc = new RTCPeerConnection({ iceServers: ICE_SERVERS });
   437→   437→      this.pc = pc;
   438→   438→      this.remoteStream = new MediaStream();
   439→   439→
   440→   440→      pc.ontrack = (ev) => {
   441→   441→        for (const track of ev.streams[0]?.getTracks() ?? [ev.track]) {
   442→   442→          if (!this.remoteStream?.getTracks().some((t) => t.id === track.id)) {
   443→   443→            this.remoteStream?.addTrack(track);
   444→   444→          }
   445→   445→        }
   446→   446→        useRealcallStore.getState().setRemoteStream(this.remoteStream);
   447→   447→      };
   448→   448→      pc.onicecandidate = (ev) => {
   449→   449→        if (ev.candidate && this.call) {
   450→   450→          this.emit("call:ice", { callId: this.call.callId, candidate: ev.candidate.toJSON() });
   451→   451→        }
   452→   452→      };
   453→   453→      pc.onconnectionstatechange = () => {
   454→   454→        if (!this.call) return;
   455→   455→        if (pc.connectionState === "connected") {
   456→   456→          this.setPhase("active");
   457→   457→          if (this.call.startedAt === null) {
   458→   458→            this.call = { ...this.call, startedAt: Date.now() };
   459→   459→            useRealcallStore.getState().setCall(this.call);
   460→   460→          }
   461→   461→        } else if (pc.connectionState === "failed") {
   462→   462→          this.failCall("Koneksi media gagal (jaringan)");
   463→   463→        }
   464→   464→      };
   465→   465→      resolve();
   466→   466→    });
   467→   467→  }

   469→   469→  private setPhase(phase: CallPhase): void {
   470→   470→    if (!this.call) return;
   471→   471→    this.call = { ...this.call, phase };
   472→   472→    useRealcallStore.getState().setCall(this.call);
   473→   473→  }

   533→   533→  async joinConf(roomId: string): Promise<{ ok: boolean; error?: string }> {
   534→   534→    if (!this.socket?.connected) return { ok: false, error: "Tidak terhubung ke server realtime" };
   535→   535→    if (this.call) return { ok: false, error: "Akhiri panggilan dahulu" };
   536→   536→    try {
   537→   537→      this.confLocalStream = await navigator.mediaDevices.getUserMedia({ audio: true, video: { width: { ideal: 1280 } } });
   538→   538→    } catch {
   539→   539→      try {
   540→   540→        this.confLocalStream = await navigator.mediaDevices.getUserMedia({ audio: true });
   541→   541→      } catch {
   542→   542→        return { ok: false, error: "Mikrofon tidak dapat diakses" };
   543→   543→      }
   544→   544→    }
   545→   545→    useRealcallStore.getState().setConfLocalStream(this.confLocalStream);
   546→   546→
   547→   547→    return new Promise((resolve) => {
   548→   548→      const s = this.socket!;
   549→   549→      const timeout = setTimeout(() => resolve({ ok: false, error: "Server tidak merespon" }), 8000);
   550→   550→      s.emit("conf:join", { roomId }, async (res: { ok: boolean; peers?: Array<{ peerSocketId: string; userId: string; username: string; accessId: string }> }) => {
   551→   551→        clearTimeout(timeout);
   552→   552→        if (!res?.ok) return resolve({ ok: false, error: "Gagal bergabung ruangan" });
   553→   553→        this.confRoomId = roomId;
   554→   554→        useRealcallStore.getState().setConf(roomId, true);
   555→   555→        // Peer yang SUDAH di ruang membuat offer ke pendatang → kita (pendatang)
   556→   556→        // hanya menyiapkan dan menunggu offer mereka.
   557→   557→        resolve({ ok: true });
   558→   558→      });
   559→   559→    });
   560→   560→  }

   602→   602→  /** Peer lama → buat offer ke pendatang baru. */
   603→   603→  private async confCreateOfferTo(peerSocketId: string, user: { userId: string; username: string; accessId: string }): Promise<void> {
   604→   604→    if (!this.confRoomId || this.confPeers.has(peerSocketId)) return;
   605→   605→    const local = await this.confEnsureLocal();
   606→   606→    if (!local) return;
   607→   607→    const pc = new RTCPeerConnection({ iceServers: ICE_SERVERS });
   608→   608→    this.confPeers.set(peerSocketId, pc);
   609→   609→    const remote = new MediaStream();
   610→   610→    this.confStreams.set(peerSocketId, remote);
   611→   611→
   612→   612→    for (const t of local.getTracks()) pc.addTrack(t, local);
   613→   613→    pc.ontrack = (ev) => {
   614→   614→      for (const track of ev.streams[0]?.getTracks() ?? [ev.track]) {
   615→   615→        if (!remote.getTracks().some((t) => t.id === track.id)) remote.addTrack(track);
   616→   616→      }
   617→   617→      useRealcallStore.getState().setConfStream(peerSocketId, remote, user);
   618→   618→    };
   619→   619→    pc.onicecandidate = (ev) => {
   620→   620→      if (ev.candidate) {
   621→   621→        this.emit("conf:relay", { roomId: this.confRoomId, toPeer: peerSocketId, kind: "ice", payload: ev.candidate.toJSON() });
   622→   622→      }
   623→   623→    };
   624→   624→    try {
   625→   625→      const offer = await pc.createOffer();
   626→   626→      await pc.setLocalDescription(offer);
   627→   627→      this.emit("conf:relay", { roomId: this.confRoomId, toPeer: peerSocketId, kind: "offer", payload: pc.localDescription });
   628→   628→      useRealcallStore.getState().addConfPeer(peerSocketId, user);
   629→   629→    } catch {
   630→   630→      this.confClosePeer(peerSocketId);
   631→   631→    }
   632→   632→  }

   634→   634→  /** Pendatang → proses offer/answer/ice dari peer lama. */
   635→   635→  private async confHandleRelay(p: { fromPeer: string; fromUser?: { userId: string; username: string }; kind: "offer" | "answer" | "ice"; payload: unknown }): Promise<void> {
   636→   636→    if (!this.confRoomId) return;
   637→   637→    let pc = this.confPeers.get(p.fromPeer);
   638→   638→    if (p.kind === "offer") {
   639→   639→      if (!pc) {
   640→   640→        const local = await this.confEnsureLocal();
   641→   641→        if (!local) return;
   642→   642→        pc = new RTCPeerConnection({ iceServers: ICE_SERVERS });
   643→   643→        this.confPeers.set(p.fromPeer, pc);
   644→   644→        const remote = new MediaStream();
   645→   645→        this.confStreams.set(p.fromPeer, remote);
   646→   646→        for (const t of local.getTracks()) pc.addTrack(t, local);
   647→   647→        pc.ontrack = (ev) => {
   648→   648→          for (const track of ev.streams[0]?.getTracks() ?? [ev.track]) {
   649→   649→            if (!remote.getTracks().some((t) => t.id === track.id)) remote.addTrack(track);
   650→   650→          }
   651→   651→          useRealcallStore.getState().setConfStream(p.fromPeer, remote, {
   652→   652→            userId: p.fromUser?.userId ?? "?",
   653→   653→            username: p.fromUser?.username ?? "Peserta",
   654→   654→            accessId: "",
   655→   655→          });
   656→   656→        };
   657→   657→        pc.onicecandidate = (ev) => {
   658→   658→          if (ev.candidate) {
   659→   659→            this.emit("conf:relay", { roomId: this.confRoomId, toPeer: p.fromPeer, kind: "ice", payload: ev.candidate.toJSON() });
   660→   660→          }
   661→   661→        };
   662→   662→        useRealcallStore.getState().addConfPeer(p.fromPeer, {
   663→   663→          userId: p.fromUser?.userId ?? "?",
   664→   664→          username: p.fromUser?.username ?? "Peserta",
   665→   665→          accessId: "",
   666→   666→        });
   667→   667→      }
   668→   668→      try {
   669→   669→        await pc.setRemoteDescription(new RTCSessionDescription(p.payload as RTCSessionDescriptionInit));
   670→   670→        const answer = await pc.createAnswer();
   671→   671→        await pc.setLocalDescription(answer);
   672→   672→        this.emit("conf:relay", { roomId: this.confRoomId, toPeer: p.fromPeer, kind: "answer", payload: pc.localDescription });
   673→   673→      } catch {
   674→   674→        this.confClosePeer(p.fromPeer);
   675→   675→      }
   676→   676→      return;
   677→   677→    }
   678→   678→    if (!pc) return;
   679→   679→    try {
   680→   680→      if (p.kind === "answer") {
   681→   681→        await pc.setRemoteDescription(new RTCSessionDescription(p.payload as RTCSessionDescriptionInit));
   682→   682→      } else {
   683→   683→        await pc.addIceCandidate(new RTCIceCandidate(p.payload as RTCIceCandidateInit)).catch(() => undefined);
   684→   684→      }
   685→   685→    } catch {
   686→   686→      // kandidat/answer usang — abaikan
   687→   687→    }
   688→   688→  }
   689→   689→
   690→   690→  private confClosePeer(peerSocketId: string): void {
   691→   691→    const pc = this.confPeers.get(peerSocketId);
   692→   692→    if (pc) {
   693→   693→      try {
   694→   694→        pc.close();
   695→   695→      } catch {
   696→   696→        // abaikan
   697→   697→      }
   698→   698→    }
   699→   699→    this.confPeers.delete(peerSocketId);
   700→   700→    this.confStreams.delete(peerSocketId);
   701→   701→  }

   704→   704→/** Warna avatar stabil dari Access ID (mis. ID-000001 → kunci palet). */
   705→   705→export function colorFromAccessId(accessId: string): string {
   706→   706→  const keys = ["emerald", "amber", "violet", "rose", "teal", "orange", "cyan", "lime"];
   707→   707→  let h = 0;
   708→   708→  for (let i = 0; i < accessId.length; i++) h = (h * 31 + accessId.charCodeAt(i)) >>> 0;
   709→   709→  return keys[h % keys.length];
   710→   710→}
