   428→      window.sessionStorage.removeItem("wa_locked");

   438→      window.sessionStorage.setItem("wa_locked", "1");

   452→      // abaikan
   453→    }

   493→          <Button onClick={unlockApp} className="w-full min-h-11" aria-label="Buka kunci aplikasi">
