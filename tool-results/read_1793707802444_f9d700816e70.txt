   365→        type="button"
   366→        onClick={() => setTheme(isDark ? "light" : "dark")}
   367→        className="w-full flex items-center gap-3 px-4 py-2.5 text-sm text-left transition-colors hover:bg-secondary"
   368→        role="menuitemcheckbox"
   369→        aria-checked={isDark}
   370→      >
   371→        <Moon className={cn("h-[18px] w-[18px] shrink-0", isDark ? "text-primary" : "text-muted-foreground")} />
   372→        <span>Mode Gelap</span>
   373→        {isDark && <Check className="ml-auto h-4 w-4 text-primary" aria-label="Aktif" />}
   374→      </button>
   375→      <button
   376→        type="button"
   377→        onClick={() => setLite(!lite)}

   492→/** Cincin progres MINI utk pojok panel header Rutinitas (mobile) — pasangan ringkap dari ProgressRing di RoutinesTab. */
   493→function MiniProgressRing({ value, max }: { value: number; max: number }) {
   494→  const r = 15;
   495→  const c = 2 * Math.PI * r;
   496→  const pct = max > 0 ? Math.min(1, value / max) : 0;
   497→  return (

   501→        <circle
   502→          cx="18"
   503→          cy="18"
   504→          r={r}
   505→          fill="none"
   506→          strokeWidth="4"
   507→          strokeLinecap="round"
   508→          className="stroke-maroon-500 transition-all duration-500"
   509→          strokeDasharray={c}
   510→          strokeDashoffset={c * (1 - pct)}
   511→        />
   512→      </svg>
   513→      <div className="absolute inset-0 flex flex-col items-center justify-center leading-none">
   514→        <span className="text-[10px] font-bold">{value}</span>

   517→    </div>
