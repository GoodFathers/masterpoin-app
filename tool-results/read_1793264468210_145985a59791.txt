   120→   120→/** Cincin progres kecil "x/y hari ini". */
   121→   121→function ProgressRing({ value, max }: { value: number; max: number }) {
   122→   122→  const r = 26;
   123→   123→  const c = 2 * Math.PI * r;
   124→   124→  const pct = max > 0 ? Math.min(1, value / max) : 0;
   125→   125→  return (
   126→   126→    <div className="relative h-16 w-16 shrink-0" role="img" aria-label={`${value} dari ${max} rutinitas dan kegiatan selesai hari ini`}>
   127→   127→      <svg viewBox="0 0 64 64" className="h-16 w-16 -rotate-90">
   128→   128→        <circle cx="32" cy="32" r={r} fill="none" strokeWidth="7" className="stroke-emerald-100 dark:stroke-emerald-950" />
   129→   129→        <circle
   130→   130→          cx="32"
   131→   131→          cy="32"
   132→   132→          r={r}
   133→   133→          fill="none"
   134→   134→          strokeWidth="7"
   135→   135→          strokeLinecap="round"
   136→   136→          className="stroke-emerald-500 transition-all duration-500"
   137→   137→          strokeDasharray={c}
   138→   138→          strokeDashoffset={c * (1 - pct)}
   139→   139→        />
   140→   140→      </svg>
   141→   141→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   142→   142→        <span className="text-sm font-bold leading-none">{value}</span>
   143→   143→        <span className="text-[10px] text-muted-foreground leading-none mt-0.5">/{max}</span>
   144→   144→      </div>
   145→   145→    </div>
   146→   146→  );
   147→   147→}

   149→   149→/** Satu sel pada papan Check-In 30 Hari. */
   150→   150→function CheckInCell({
   151→   151→  pos,
   152→   152→  done,
   153→   153→  isNext,
   154→   154→  checkedToday,
   155→   155→}: {
   156→   156→  pos: number;
   157→   157→  done: boolean;
   158→   158→  isNext: boolean;
   159→   159→  checkedToday: boolean;
   160→   160→}) {
   161→   161→  const milestone = isCheckInMilestone(pos);
   162→   162→  const base =
   163→   163→    "relative flex aspect-square flex-col items-center justify-center rounded-xl border text-[10px] font-semibold transition-colors";
   164→   164→  const cls = done
   165→   165→    ? milestone
   166→   166→      ? "border-amber-300 bg-amber-400/90 text-amber-950 shadow-sm"
   167→   167→      : "border-emerald-600 bg-emerald-600 text-white shadow-sm"
   168→   168→    : isNext
   169→   169→      ? "border-2 border-emerald-500 bg-emerald-50 text-emerald-700 dark:bg-emerald-950/60 dark:text-emerald-300 animate-pulse"
   170→   170→      : "border-border bg-muted/40 text-muted-foreground/70";
   171→   171→
   172→   172→  return (
   173→   173→    <div className={`${base} ${cls}`} title={`Hari ${pos}${milestone ? ` — milestone +${pos === 30 ? "20.000" : pos === 21 ? "7.500" : pos === 14 ? "5.000" : "3.000"} mP` : ""}`}>
   174→   174→      {done ? (
   175→   175→        milestone ? (
   176→   176→          <Gift className="h-4 w-4" aria-hidden="true" />
   177→   177→        ) : (
   178→   178→          <Check className="h-3.5 w-3.5" aria-hidden="true" />
   179→   179→        )
   180→   180→      ) : milestone ? (
   181→   181→        <Gift className="h-3.5 w-3.5 opacity-80" aria-hidden="true" />
   182→   182→      ) : (
   183→   183→        <span>{pos}</span>
   184→   184→      )}
   185→   185→      {milestone && (
   186→   186→        <span className="text-[8px] leading-none mt-0.5 font-bold">{pos === 30 ? "20rb" : pos === 21 ? "7,5rb" : pos === 14 ? "5rb" : "3rb"}</span>
   187→   187→      )}
   188→   188→      {isNext && !checkedToday && (
   189→   189→        <span className="absolute -top-1 -right-1 h-3 w-3 rounded-full bg-emerald-500 ring-2 ring-background" aria-hidden="true" />
   190→   190→      )}
   191→   191→      <span className="sr-only">{`Hari ${pos}${done ? " sudah check-in" : milestone ? " milestone" : ""}`}</span>
   192→   192→    </div>
   193→   193→  );
   194→   194→}
