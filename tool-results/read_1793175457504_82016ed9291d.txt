    99→  // ---------- Fitur menu baru ----------
   100→  const [searchOpen, setSearchOpen] = useState(false);

   102→  const [selectMode, setSelectMode] = useState(false);

   116→  const scrollRef = useRef<HTMLDivElement | null>(null);
   117→  const atBottomRef = useRef(true);
   118→  // Ref pesan terkini untuk dipakai di dalam listener socket (tanpa re-subscribe)
   119→  const messagesRef = useRef<ChatMessage[]>([]);
   120→  messagesRef.current = messages;
   121→
   122→  // ---------- Muat pesan (paginasi Batch 4) ----------
   123→  // Muat jendela TERBARU (60 pesan). Riwayat lebih lama dimuat bertahap saat
   124→  // gulir ke atas (loadOlder), sinkronisasi berkala hanya ambil pesan baru
   125→  // (syncTail) — tidak lagi mengunduh seluruh isi obrolan tiap kali.
   126→  const loadMessages = useCallback(
   127→    async (silent = false) => {
   128→      if (!silent) setLoading(true);
   129→      try {
   130→        const res = await fetch(`/api/chats/${chat.id}/messages?limit=${PAGE_SIZE}`, { cache: "no-store" });
   131→        const data = (await res.json()) as {
   132→          messages?: ChatMessage[];
   133→          participants?: Participant[];
   134→          hasMore?: boolean;
   135→          error?: string;
   136→        };
   137→        if (res.ok) {
   138→          setMessages(data.messages ?? []);
   139→          setHasMore(!!data.hasMore);
   140→          if (data.participants) setParticipants(data.participants);
   141→        }
   142→      } catch {
   143→        if (!silent) toast({ title: "Gagal memuat pesan", variant: "destructive" });
   144→      } finally {
   145→        if (!silent) setLoading(false);
   146→      }
   147→    },
   148→    [chat.id, toast]
   149→  );
   150→
   151→  // ---------- Sinkron ekor: hanya pesan yang lebih baru dari yang sudah termuat ----------
   152→  const syncTail = useCallback(async () => {
   153→    const current = messagesRef.current;
   154→    const lastReal = [...current].reverse().find((m) => !m.id.startsWith("tmp-"));
   155→    if (!lastReal) {
   156→      void loadMessages();
   157→      return;
   158→    }
   159→    try {
   160→      const res = await fetch(
   161→        `/api/chats/${chat.id}/messages?after=${encodeURIComponent(lastReal.createdAt)}`,
   162→        { cache: "no-store" }
   163→      );
   164→      const data = (await res.json()) as { messages?: ChatMessage[] };
   165→      if (!res.ok || !data.messages?.length) return;
   166→      setMessages((prev) => {
   167→        const known = new Set(prev.map((m) => m.id));
   168→        const fresh = data.messages!.filter((m) => !known.has(m.id));
   169→        return fresh.length ? [...prev, ...fresh] : prev;
   170→      });
   171→    } catch {
   172→      // fallback senyap — polling berikutnya mencoba lagi
   173→    }
   174→  }, [chat.id, loadMessages]);
   175→
   176→  // ---------- Muat riwayat lebih lama saat gulir ke atas ----------
   177→  const loadOlder = useCallback(async () => {
   178→    if (loadingOlderRef.current) return;
   179→    const oldest = messagesRef.current.find((m) => !m.id.startsWith("tmp-"));
   180→    if (!oldest) return;
   181→    loadingOlderRef.current = true;
   182→    setLoadingOlder(true);
   183→    const el = scrollRef.current;
   184→    const prevHeight = el?.scrollHeight ?? 0;
   185→    const prevTop = el?.scrollTop ?? 0;
   186→    try {
   187→      const res = await fetch(
   188→        `/api/chats/${chat.id}/messages?before=${encodeURIComponent(oldest.createdAt)}&limit=${PAGE_SIZE}`,
   189→        { cache: "no-store" }
   190→      );
   191→      const data = (await res.json()) as { messages?: ChatMessage[]; hasMore?: boolean };
   192→      if (res.ok) {
   193→        setHasMore(!!data.hasMore);
   194→        if (data.messages?.length) {
   195→          setMessages((prev) => {
   196→            const known = new Set(prev.map((m) => m.id));
   197→            const older = data.messages!.filter((m) => !known.has(m.id));
   198→            if (!older.length) return prev;
   199→            return [...older, ...prev];
   200→          });
   201→          // Pertahankan posisi gulir setelah riwayat lama disisipkan di atas
   202→          requestAnimationFrame(() => {
   203→            const el2 = scrollRef.current;
   204→            if (el2) el2.scrollTop = el2.scrollHeight - prevHeight + prevTop;
   205→          });
   206→        }
   207→      }
   208→    } catch {
   209→      // senyap — pengguna bisa mencoba gulir lagi
   210→    } finally {
   211→      loadingOlderRef.current = false;
   212→      setLoadingOlder(false);
   213→    }
   214→  }, [chat.id]);
   215→
   216→  useEffect(() => {
   217→    atBottomRef.current = true;
   218→    void loadMessages();
   219→  }, [loadMessages]);
   220→
   221→  // Tema latar tersimpan per chat
   222→  useEffect(() => {

   234→  // Tandai sudah dibaca saat chat dibuka & saat pesan baru masuk ke chat

   236→  useEffect(() => {

   239→        method: "PATCH",

   242→      }).then(() => onChatUpdated());
   243→    }
   244→  }, [chat.id, chat.unread]);
   245→
   246→  // Daftarkan chat aktif global — NotificationCenter melewatkan notifikasi
   247→  // untuk chat yang sedang terbuka di tab ini.
   248→  useEffect(() => {
   249→    setActiveChatId(chat.id);
   250→    return () => setActiveChatId(null);
   251→  }, [chat.id]);
   252→
   253→  // Polling pesan setiap 15 detik (dilewati saat ada pesan optimist) —
   254→  // fallback bila push realtime gagal/terputus. Hanya mengambil pesan BARU,
   255→  // bukan seluruh obrolan (Batch 4).
   256→  useEffect(() => {
   257→    const t = setInterval(() => {
   258→      const hasPending = messagesRef.current.some((m) => m.id.startsWith("tmp-"));
   259→      if (!hasPending) void syncTail();
   260→    }, 15_000);
   261→    return () => clearInterval(t);
   262→  }, [syncTail]);
   263→
   264→  // ---------- Realtime (Batch 2): push socket.io ----------
   265→  // Pesan masuk, status centang (DELIVERED/READ), hapus pesan, dan sinkron
   266→  // isi obrolan dari tab/perangkat lain — tanpa menunggu polling.
   267→  useEffect(() => {
   268→    const s = getRealtimeSocket();
   269→
   270→    const onMessage = (p: ChatMessagePush) => {
   271→      if (!p || p.chatId !== chat.id || !p.message) return;
   272→      setMessages((prev) =>
   273→        prev.some((m) => m.id === p.message.id) ? prev : [...prev, p.message]
   274→      );
   275→      if (p.message.sender !== "me") atBottomRef.current = true;
   276→    };
   277→
   278→    const onStatus = (p: ChatStatusPush) => {
   279→      if (!p || p.chatId !== chat.id) return;
   280→      setMessages((prev) =>
   281→        prev.map((m) =>
   282→          m.sender === "me" && !m.id.startsWith("tmp-") ? { ...m, status: p.status } : m
   283→        )
   284→      );
   285→    };
   286→
   287→    const onMessageDeleted = (p: ChatMessageDeletedPush) => {
   288→      if (!p || p.chatId !== chat.id) return;
   289→      setMessages((prev) => prev.filter((m) => m.id !== p.messageId));
   290→    };
   291→
   292→    const onUpdate = (p: ChatUpdatePush) => {
   293→      if (!p || p.chatId !== chat.id) return;
   294→      const hasPending = messagesRef.current.some((m) => m.id.startsWith("tmp-"));
   295→      if (!hasPending) void syncTail();
   296→    };
   297→
   298→    s.on("chat:message", onMessage);
   299→    s.on("chat:status", onStatus);
   300→    s.on("chat:message-deleted", onMessageDeleted);
   301→    s.on("chat:update", onUpdate);
   302→    return () => {
   303→      s.off("chat:message", onMessage);
   304→      s.off("chat:status", onStatus);
   305→      s.off("chat:message-deleted", onMessageDeleted);
   306→      s.off("chat:update", onUpdate);
   307→    };
   308→  }, [chat.id, syncTail]);
   309→
   310→  // Auto-scroll ke bawah
   311→  useEffect(() => {
   312→    const el = scrollRef.current;
   313→    if (!el) return;
   314→    const last = messages[messages.length - 1];
   315→    if (atBottomRef.current || last?.sender === "me") {
   316→      el.scrollTop = el.scrollHeight;
   317→    }
   318→  }, [messages]);
   319→
   320→  // Data produk untuk kartu PRODUCT
   321→  useEffect(() => {
   322→    if (productsMap.size > 0) return;
   323→    if (!messages.some((m) => m.type === "PRODUCT")) return;
   324→    fetch("/api/products", { cache: "no-store" })
   325→      .then((r) => r.json())
   326→      .then((d: { products?: ProductRow[] }) => {
   327→        setProductsMap(new Map((d.products ?? []).map((p) => [p.id, p])));
   328→      })
   329→      .catch(() => setProductsMap(new Map()));
   330→  }, [messages, productsMap.size]);
   331→
   332→  const handleScroll = () => {
   333→    const el = scrollRef.current;
   334→    if (!el) return;
   335→    atBottomRef.current = el.scrollHeight - el.scrollTop - el.clientHeight < 120;
   336→    // Gulir mendekati atas → muat riwayat lebih lama (paginasi Batch 4)
   337→    if (el.scrollTop < 80 && hasMore && !loadingOlder) void loadOlder();
   338→  };
   339→
   340→  // ---------- Kirim pesan (alur WA: PENDING → SENT → DELIVERED → balasan) ----------
   341→  const sendMessage = useCallback(
   342→    (payload: Partial<ChatMessage> & { type: string }) => {
   343→      const tempId = `tmp-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   344→      const optimistic: ChatMessage = {
   345→        id: tempId,
   346→        chatId: chat.id,
   347→        sender: "me",
   348→        senderName: "Anda",
   349→        senderColorKey: "emerald",
   350→        type: payload.type,
   351→        body: payload.body ?? "",
   352→        mediaUrl: payload.mediaUrl ?? null,
   353→        mediaName: payload.mediaName ?? null,
   354→        mediaMime: payload.mediaMime ?? null,
   355→        mediaSize: payload.mediaSize ?? null,
   356→        latitude: payload.latitude ?? null,
   357→        longitude: payload.longitude ?? null,
   358→        placeName: payload.placeName ?? null,
   359→        cardName: payload.cardName ?? null,
   360→        cardPhone: payload.cardPhone ?? null,
   361→        productId: payload.productId ?? null,
   362→        status: "PENDING",
   363→        createdAt: new Date().toISOString(),
   364→      };
   365→      atBottomRef.current = true;
   366→      setMessages((prev) => [...prev, optimistic]);
   367→
   368→      void (async () => {
   369→        try {
   370→          const res = await fetch(`/api/chats/${chat.id}/messages`, {
   371→            method: "POST",
   372→            headers: { "Content-Type": "application/json" },
   373→            body: JSON.stringify(payload),
   374→          });
   375→          const data = (await res.json()) as { message?: ChatMessage; error?: string };
   376→          if (!res.ok || !data.message) throw new Error(data.error || "Gagal mengirim pesan");
   377→          const serverMsg = data.message;
   378→          setMessages((prev) => prev.map((m) => (m.id === tempId ? serverMsg : m)));
   379→          onChatUpdated();
   380→
   381→          // Simulasi pesan tersampaikan
   382→          setTimeout(() => {
   383→            void fetch("/api/simulate/progress", {
   384→              method: "POST",
   385→              headers: { "Content-Type": "application/json" },
   386→              body: JSON.stringify({ chatId: chat.id }),
   387→            })
   388→              .then(() => {
   389→                setMessages((prev) => prev.map((m) => (m.id === serverMsg.id ? { ...m, status: "DELIVERED" } : m)));
   390→              })
   391→              .catch(() => {});
   392→          }, 900);
   393→
   394→          // Simulasi balasan kontak
   395→          setTimeout(() => {
   396→            void fetch("/api/simulate/reply", {
   397→              method: "POST",
   398→              headers: { "Content-Type": "application/json" },
   399→              body: JSON.stringify({ chatId: chat.id }),
   400→            })
   401→              .then((r) => r.json())
   402→              .then((d: { message?: ChatMessage }) => {
   403→                if (d.message) {
   404→                  setMessages((prev) =>
   405→                    prev.some((x) => x.id === d.message!.id) ? prev : [...prev, d.message!]
   406→                  );
   407→                  atBottomRef.current = true;
   408→                  onChatUpdated();
   409→                }
   410→              })
   411→              .catch(() => {});
   412→          }, 2500 + Math.floor(Math.random() * 2500));
   413→        } catch (err) {
   414→          setMessages((prev) => prev.map((m) => (m.id === tempId ? { ...m, status: "FAILED" } : m)));
   415→          toast({
   416→            title: "Pesan gagal dikirim",
   417→            description: err instanceof Error ? err.message : "Periksa koneksi lalu coba lagi.",
   418→            variant: "destructive",
   419→          });
   420→        }
   421→      })();
   422→    },
   423→    [chat.id, onChatUpdated, toast]
   424→  );

   466→      } catch {
   467→        toast({ title: "Gagal memperbarui obrolan", variant: "destructive" });
   468→      }
   469→    },
   470→    [chat.id, onChatUpdated, toast]
   471→  );
   472→
   473→  const muteChat = (durationLabel: string) => {

   475→  };
   476→
   477→  const saveContactName = async () => {

   489→      setContactEditOpen(false);
   490→      toast({ title: "Kontak disimpan", description: name });
   491→      onChatUpdated();
   492→    } catch {
   493→      toast({ title: "Gagal menyimpan kontak", variant: "destructive" });
   494→    }
   495→  };
   496→
   497→  const exportChat = () => {
   498→    const lines = messages.map((m) => {
   499→      const who = m.sender === "me" ? "Anda" : m.sender === "system" ? "Sistem" : m.senderName ?? chat.name;
   500→      const text = m.body || `[${m.type.toLowerCase()}]`;
   501→      return `[${waDayLabel(m.createdAt)} ${formatTime(m.createdAt)}] ${who}: ${text}`;
   502→    });
   503→    const content =
   504→      `Obrolan dengan ${chat.name}\n` +
   505→      `Diekspor dari ${business.businessName} · ${new Date().toLocaleString("id-ID")}\n` +
   506→      `${"-".repeat(40)}\n\n${lines.join("\n")}\n`;
   507→    const blob = new Blob([content], { type: "text/plain;charset=utf-8" });
   508→    const url = URL.createObjectURL(blob);
   509→    const a = document.createElement("a");
   510→    a.href = url;
   511→    a.download = `Obrolan ${chat.name}.txt`;
   512→    a.click();
   513→    URL.revokeObjectURL(url);
   514→    toast({ title: "Obrolan diekspor", description: "Tersimpan sebagai berkas .txt" });
   515→  };
   516→
   517→  const clearMessages = async () => {
   518→    try {
   519→      const res = await fetch(`/api/chats/${chat.id}/messages`, { method: "DELETE" });
   520→      if (!res.ok) throw new Error();
   521→      setMessages([]);
   522→      onChatUpdated();
   523→      toast({ title: "Obrolan dibersihkan" });
   524→    } catch {
   525→      toast({ title: "Gagal membersihkan obrolan", variant: "destructive" });
   526→    }
   527→  };
   528→
   529→  const deleteChat = async () => {
   530→    try {
   531→      const res = await fetch(`/api/chats/${chat.id}`, { method: "DELETE" });
