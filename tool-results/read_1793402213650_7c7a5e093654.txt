    14→function tinyWav(seconds = 1.2, freq = 620): Buffer {
    15→  const sampleRate = 8000;
    16→  const total = Math.floor(sampleRate * seconds);
    17→  const data = Buffer.alloc(total * 2);
    18→  for (let i = 0; i < total; i++) {
    19→    const t = i / sampleRate;
    20→    const fade = Math.min(1, t * 8) * Math.min(1, (seconds - t) * 4);
    21→    const v = Math.sin(2 * Math.PI * freq * t) * 0.25 * fade;
    22→    data.writeInt16LE(Math.round(v * 32767), i * 2);
    23→  }
    24→  const header = Buffer.alloc(44);
    25→  header.write("RIFF", 0);
    26→  header.writeUInt32LE(36 + data.length, 4);
    27→  header.write("WAVE", 8);
    28→  header.write("fmt ", 12);
    29→  header.writeUInt32LE(16, 16);
    30→  header.writeUInt16LE(1, 20);
    31→  header.writeUInt16LE(1, 22);
    32→  header.writeUInt32LE(sampleRate, 24);
    33→  header.writeUInt32LE(sampleRate * 2, 28);
    34→  header.writeUInt16LE(2, 32);
    35→  header.writeUInt16LE(16, 34);
    36→  header.write("data", 36);
    37→  header.writeUInt32LE(data.length, 40);
    38→  return Buffer.concat([header, data]);
    39→}

    41→// PDF satu halaman minimal (Batch 4: dokumen seed adalah berkas nyata di disk,

    43→function tinyPdf(): Buffer {
    44→  const lines = [
    45→    "Invoice BJ-2024-0187",
    46→    "IndiGoo - Toko Souvenir & Pangan Olahan",
    47→    "",
    48→    "Batik Parang Sogan (2 m) x 2    500.000 mP",
    49→    "Total: 500.000 mP",
    50→    "",
    51→    "Terima kasih atas pesanan Anda.",
    52→  ];
    53→  let content = "";
    54→  let y = 790;
    55→  for (const ln of lines) {
    56→    content += `BT /F1 12 Tf 56 ${y} Td (${ln.replace(/[\\()]/g, " ")}) Tj ET\n`;
    57→    y -= 24;
    58→  }
    59→  const objects = [
    60→    "<< /Type /Catalog /Pages 2 0 R >>",
    61→    "<< /Type /Pages /Kids [3 0 R] /Count 1 >>",
    62→    "<< /Type /Page /Parent 2 0 R /MediaBox [0 0 595 842] /Resources << /Font << /F1 5 0 R >> >> /Contents 4 0 R >>",
    63→    `<< /Length ${Buffer.byteLength(content)} >>\nstream\n${content}endstream`,
    64→    "<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>",
    65→  ];
    66→  let pdf = "%PDF-1.4\n";
    67→  const offsets: number[] = [];
    68→  objects.forEach((body, i) => {
    69→    offsets.push(Buffer.byteLength(pdf));
    70→    pdf += `${i + 1} 0 obj\n${body}\nendobj\n`;
    71→  });
    72→  const xrefStart = Buffer.byteLength(pdf);
    73→  pdf += `xref\n0 ${objects.length + 1}\n0000000000 65535 f \n`;
    74→  for (const off of offsets) pdf += `${off.toString().padStart(10, "0")} 00000 n \n`;
    75→  pdf += `trailer\n<< /Size ${objects.length + 1} /Root 1 0 R >>\nstartxref\n${xrefStart}\n%%EOF`;
    76→  return Buffer.from(pdf, "latin1");
    77→}
