   111→/** Cincin progres kecil "x/y hari ini". */
   112→function ProgressRing({ value, max }: { value: number; max: number }) {
   113→  const r = 26;
   114→  const c = 2 * Math.PI * r;
   115→  const pct = max > 0 ? Math.min(1, value / max) : 0;
   116→  return (
   117→    <div className="relative h-16 w-16 shrink-0" role="img" aria-label={`${value} dari ${max} rutinitas dan kegiatan selesai hari ini`}>
   118→      <svg viewBox="0 0 64 64" className="h-16 w-16 -rotate-90">
   119→        <circle cx="32" cy="32" r={r} fill="none" strokeWidth="7" className="stroke-emerald-100 dark:stroke-emerald-950" />
   120→        <circle
   121→          cx="32"
   122→          cy="32"
   123→          r={r}
   124→          fill="none"
   125→          strokeWidth="7"
   126→          strokeLinecap="round"
   127→          className="stroke-emerald-500 transition-all duration-500"
   128→          strokeDasharray={c}
   129→          strokeDashoffset={c * (1 - pct)}
   130→        />
   131→      </svg>
   132→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   133→        <span className="text-sm font-bold leading-none">{value}</span>
   134→        <span className="text-[10px] text-muted-foreground leading-none mt-0.5">/{max}</span>
   135→      </div>
   136→    </div>
   137→  );
   138→}

   140→/** Satu sel pada papan Check-In 30 Hari. */
   141→function CheckInCell({
   142→  pos,
   143→  done,
   144→  isNext,
   145→  checkedToday,
   146→}: {
   147→  pos: number;
   148→  done: boolean;
   149→  isNext: boolean;
   150→  checkedToday: boolean;
   151→}) {
   152→  const milestone = isCheckInMilestone(pos);
   153→  const base =
   154→    "relative flex aspect-square flex-col items-center justify-center rounded-xl border text-[10px] font-semibold transition-colors";
   155→  const cls = done
   156→    ? milestone
   157→      ? "border-amber-300 bg-amber-400/90 text-amber-950 shadow-sm"
   158→      : "border-emerald-600 bg-emerald-600 text-white shadow-sm"
   159→    : isNext
   160→      ? "border-2 border-emerald-500 bg-emerald-50 text-emerald-700 dark:bg-emerald-950/60 dark:text-emerald-300 animate-pulse"
   161→      : "border-border bg-muted/40 text-muted-foreground/70";
   162→
   163→  return (
   164→    <div className={`${base} ${cls}`} title={`Hari ${pos}${milestone ? ` — milestone +${pos === 30 ? "20.000" : pos === 21 ? "7.500" : pos === 14 ? "5.000" : "3.000"} mP` : ""}`}>
   165→      {done ? (
   166→        milestone ? (
   167→          <Gift className="h-4 w-4" aria-hidden="true" />
   168→        ) : (
   169→          <Check className="h-3.5 w-3.5" aria-hidden="true" />
   170→        )
   171→      ) : milestone ? (
   172→        <Gift className="h-3.5 w-3.5 opacity-80" aria-hidden="true" />
   173→      ) : (
   174→        <span>{pos}</span>
   175→      )}
   176→      {milestone && (
   177→        <span className="text-[8px] leading-none mt-0.5 font-bold">{pos === 30 ? "20rb" : pos === 21 ? "7,5rb" : pos === 14 ? "5rb" : "3rb"}</span>
   178→      )}
   179→      {isNext && !checkedToday && (
   180→        <span className="absolute -top-1 -right-1 h-3 w-3 rounded-full bg-emerald-500 ring-2 ring-background" aria-hidden="true" />
   181→      )}
   182→      <span className="sr-only">{`Hari ${pos}${done ? " sudah check-in" : milestone ? " milestone" : ""}`}</span>
   183→    </div>
   184→  );
   185→}
