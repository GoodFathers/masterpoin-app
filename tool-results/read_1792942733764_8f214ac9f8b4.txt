    11→function tinyWavBase64(seconds = 1.2, freq = 620): string {
    12→  const sampleRate = 8000;
    13→  const total = Math.floor(sampleRate * seconds);
    14→  const data = Buffer.alloc(total * 2);
    15→  for (let i = 0; i < total; i++) {
    16→    const t = i / sampleRate;
    17→    const fade = Math.min(1, t * 8) * Math.min(1, (seconds - t) * 4);
    18→    const v = Math.sin(2 * Math.PI * freq * t) * 0.25 * fade;
    19→    data.writeInt16LE(Math.round(v * 32767), i * 2);
    20→  }
    21→  const header = Buffer.alloc(44);
    22→  header.write("RIFF", 0);
    23→  header.writeUInt32LE(36 + data.length, 4);
    24→  header.write("WAVE", 8);
    25→  header.write("fmt ", 12);
    26→  header.writeUInt32LE(16, 16);
    27→  header.writeUInt16LE(1, 20);
    28→  header.writeUInt16LE(1, 22);
    29→  header.writeUInt32LE(sampleRate, 24);
    30→  header.writeUInt32LE(sampleRate * 2, 28);
    31→  header.writeUInt16LE(2, 32);
    32→  header.writeUInt16LE(16, 34);
    33→  header.write("data", 36);
    34→  header.writeUInt32LE(data.length, 40);
    35→  return Buffer.concat([header, data]).toString("base64");
    36→}
